    52→    52→  const [data, setData] = useState<TubeData | null>(null);
    53→    53→  const [loadError, setLoadError] = useState(false);
    54→    54→  const [view, setView] = useState<View>("home");
    55→    55→  const [current, setCurrent] = useState<Vid | null>(null);

    72→    72→

    74→    74→  useEffect(() => {
    75→    75→    let alive = true;
    76→    76→    (async () => {
    77→    77→      await Promise.resolve();
    78→    78→      if (!alive) return;
    79→    79→      try {
    80→    80→        const h = localStorage.getItem("mytube.hist");
    81→    81→        if (h) setHist(JSON.parse(h));
    82→    82→        const a = localStorage.getItem("mytube.autoplay");
    83→    83→        if (a !== null) setAutoplay(a === "1");

    85→    85→        /* noop */
    86→    86→      }
    87→    87→      setMounted(true);
    88→    88→      try {
    89→    89→        const r = await fetch("/api/videos");

    95→    95→      }
    96→    96→    })();

    99→    99→    };

   106→   106→        localStorage.setItem("mytube.hist", JSON.stringify(next));
   107→   107→      } catch {
   108→   108→        /* noop */

   110→   110→      return next;
   111→   111→    });
   112→   112→  }, []);
   113→   113→

   115→   115→    setHist([]);
   116→   116→    try {
   117→   117→      localStorage.removeItem("mytube.hist");

   124→   124→  const changeAutoplay = useCallback((v: boolean) => {

   127→   127→      localStorage.setItem("mytube.autoplay", v ? "1" : "0");
   128→   128→    } catch {
   129→   129→      /* noop */
   130→   130→    }

   133→   133→  const channels = data?.channels ?? [];

   135→   135→  const filtered = useMemo(() => {
   136→   136→    if (!data) return [];

   139→   139→      if (tier !== "all" && (data.channels[v.c]?.tier as Tier) !== tier) return false;
   140→   140→      if (chan !== null && v.c !== chan) return false;
   141→   141→      if (ql) {
   142→   142→        const t = v.t.toLowerCase();
   143→   143→        const c = `${data.channels[v.c]?.n ?? ""} ${v.up ?? ""}`.toLowerCase();
   144→   144→        if (!t.includes(ql) && !c.includes(ql)) return false;
   145→   145→      }
   146→   146→      return true;
   147→   147→    });
   148→   148→    // Feed-level dedupe: same video via its channel AND the playlist keeps the
   149→   149→    // channel copy in the mixed feed (the ★ filter still shows the full playlist).
   150→   150→    const seen = new Set<string>();
   151→   151→    out = out.filter((v) => !seen.has(v.id) && seen.add(v.id));

   153→   153→      out = [...out].sort((a, b) => (b.ts ?? 0) - (a.ts ?? 0) || b.v - a.v);
   154→   154→    else if (sort === "views") out = [...out].sort((a, b) => b.v - a.v);
   155→   155→    else if (sort === "longest") out = [...out].sort((a, b) => b.d - a.d);
   156→   156→    else if (sort === "shuffle") out = shuffleSeeded(out, seed);
   157→   157→    return out;
   158→   158→  }, [data, q, tier, chan, sort, seed]);

   166→   166→  const openFromGrid = useCallback(

   312→   312→                        chan === i ? "bg-accent font-medium" : ""

   324→   324→          </>
   325→   325→        ) : (
   326→   326→          <div className="space-y-2 p-2">
   327→   327→            {Array.from({ length: 8 }).map((_, i) => (
   328→   328→              <Skeleton key={i} className="h-6 w-full" />
   329→   329→            ))}
   330→   330→          </div>
   331→   331→        )}
   332→   332→      </div>
   333→   333→    </div>
   334→   334→  );
   335→   335→
   336→   336→  const chip = (t: "all" | Tier) => {
   337→   337→    const active = tier === t;
   338→   338→    return (
   339→   339→      <button
   340→   340→        key={t}

   359→   359→      </button>
   360→   360→    );
