    61→    61→  const [chan, setChan] = useState<number | null>(null);
    62→    62→  const [sort, setSort] = useState<SortMode>("newest");
    63→    63→  const [seed, setSeed] = useState(1);

    65→    65→  // Ghamidi circle, SAMAA/Geo reposts, ...) from feed, search and TV queue.

    70→    70→  const [tvQueue, setTvQueue] = useState<Vid[]>([]);
    71→    71→  const [tvStart, setTvStart] = useState(0);
    72→    72→  const [hist, setHist] = useState<HistEntry[]>([]);
    73→    73→  const [autoplay, setAutoplay] = useState(true);

    77→    77→
    78→    78→  // Data + persisted prefs (async: never setState synchronously inside the effect)
    79→    79→  useEffect(() => {

    81→    81→    (async () => {
    82→    82→      await Promise.resolve();

    84→    84→      try {
    85→    85→        const h = localStorage.getItem("mytube.hist");

    87→    87→        const a = localStorage.getItem("mytube.autoplay");
    88→    88→        if (a !== null) setAutoplay(a === "1");
    89→    89→        if (localStorage.getItem("chadtube.pk") === "1") setPkMode(true);
    90→    90→      } catch {
    91→    91→        /* noop */
    92→    92→      }
    93→    93→      setMounted(true);

    95→    95→        const r = await fetch("/api/videos");

   112→   112→        localStorage.setItem("mytube.hist", JSON.stringify(next));
   113→   113→      } catch {
   114→   114→        /* noop */
   115→   115→      }
   116→   116→      return next;
   117→   117→    });
   118→   118→  }, []);
   119→   119→

   121→   121→    setHist([]);
   122→   122→    try {
   123→   123→      localStorage.removeItem("mytube.hist");

   125→   125→      /* noop */
   126→   126→    }
   127→   127→    toast({ title: "Watch history cleared" });

   134→   134→    } catch {
   135→   135→      /* noop */
   136→   136→    }
   137→   137→  }, []);

   145→   145→      if (tier !== "all" && (data.channels[v.c]?.tier as Tier) !== tier) return false;
   146→   146→      if (chan !== null && v.c !== chan) return false;
   147→   147→      if (pkMode && (data.channels[v.c]?.pk || v.pk)) return false;
   148→   148→      if (ql) {
   149→   149→        const t = v.t.toLowerCase();
   150→   150→        const c = `${data.channels[v.c]?.n ?? ""} ${v.up ?? ""}`.toLowerCase();
   151→   151→        if (!t.includes(ql) && !c.includes(ql)) return false;
   152→   152→      }
   153→   153→      return true;
   154→   154→    });

   156→   156→    // channel copy in the mixed feed (the ★ filter still shows the full playlist).

   160→   160→      // Healthy feed: newest-first inside each channel, then round-robin

   162→   162→      // cluster at the top. Single-channel filters degenerate to identity.

   164→   164→        [...out].sort((a, b) => (b.ts ?? 0) - (a.ts ?? 0) || b.v - a.v)
   165→   165→      );
   166→   166→    } else if (sort === "views") out = [...out].sort((a, b) => b.v - a.v);
   167→   167→    else if (sort === "longest") out = [...out].sort((a, b) => b.d - a.d);
   168→   168→    else if (sort === "shuffle") out = shuffleSeeded(out, seed);
   169→   169→    return out;
   170→   170→  }, [data, dq, tier, chan, sort, seed, pkMode]);

   335→   335→                        setChan(i);
   336→   336→                        applyFilter();
   337→   337→                        setView("home");
   338→   338→                        onNavigate?.();
   339→   339→                      }}
   340→   340→                      className={`flex w-full items-center justify-between gap-2 rounded-lg px-3 py-1.5 text-sm hover:bg-accent ${

   384→   384→            style={{ backgroundColor: TIER_COLORS[t] }}

   388→   388→      </button>

   410→   410→          >

   414→   414→              width={36}
   415→   415→              height={36}
   416→   416→              draggable={false}
   417→   417→              className="chad-logo h-9 w-auto select-none"
   418→   418→            />
   419→   419→            <span className="text-lg font-bold tracking-tight">ChadTube</span>
   420→   420→          </button>

   423→   423→            <Input

   431→   431→              aria-label="Search"

   435→   435→          <div className="flex items-center gap-1">
   436→   436→            <Button
