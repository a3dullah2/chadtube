     1→     1→"use client";
     2→     2→
     3→     3→import { useCallback, useEffect, useRef, useState } from "react";
     4→     4→import {
     5→     5→  ArrowLeft,
     6→     6→  Maximize,
     7→     7→  Minimize,
     8→     8→  Pause,
     9→     9→  Play,

    11→    11→  SkipForward,
    12→    12→  Volume1,

    14→    14→  VolumeX,

    18→    18→import { ScrollArea } from "@/components/ui/scroll-area";
    19→    19→import { Switch } from "@/components/ui/switch";
    20→    20→import { cn } from "@/lib/utils";

    24→    24→
    25→    25→interface WatchViewProps {
    26→    26→  video: Vid;
    27→    27→  queue: Vid[];
    28→    28→  channels: Chan[];
    29→    29→  autoplay: boolean;
    30→    30→  theater: boolean;
    31→    31→  onAutoplayChange: (v: boolean) => void;
    32→    32→  onTheaterChange: (v: boolean) => void;
    33→    33→  onBack: () => void;
    34→    34→  onSelect: (v: Vid) => void;
    35→    35→}
    36→    36→
    37→    37→export function WatchView({
    38→    38→  video,
    39→    39→  queue,
    40→    40→  channels,
    41→    41→  autoplay,
    42→    42→  theater,
    43→    43→  onAutoplayChange,
    44→    44→  onTheaterChange,
    45→    45→  onBack,
    46→    46→  onSelect,
    47→    47→}: WatchViewProps) {
    48→    48→  const idx = queue.findIndex((v) => v.id === video.id);
    49→    49→  const next =
    50→    50→    idx >= 0 && idx + 1 < queue.length ? queue[idx + 1] : (queue[0] ?? null);
    51→    51→  const chan = channels[video.c];
    52→    52→  const upNext = idx >= 0 ? queue.slice(idx + 1, idx + 26) : queue.slice(0, 25);
    53→    53→  const endRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    54→    54→  const endIntRef = useRef<ReturnType<typeof setInterval> | null>(null);
    55→    55→  const [countdown, setCountdown] = useState<number | null>(null);
    56→    56→
    57→    57→  // ----- custom player chrome (hover-only controls) -----
    58→    58→  const shellRef = useRef<HTMLDivElement | null>(null);
    59→    59→  const idleTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    60→    60→  const clickTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    61→    61→  const touchRef = useRef(false);
    62→    62→  const pstateRef = useRef(-1);
    63→    63→  const [pstate, setPstate] = useState(-1); // YT player state (-1 unstarted .. 3 buffering)
    64→    64→  const [cur, setCur] = useState(0);
    65→    65→  const [dur, setDur] = useState(0);
    66→    66→  const [buf, setBuf] = useState(0);
    67→    67→  const [muted, setMuted] = useState(false);
    68→    68→  const [vol, setVol] = useState(100);
    69→    69→  const [hover, setHover] = useState(false);
    70→    70→  const [touchUi, setTouchUi] = useState(false);
    71→    71→  const [controlsHidden, setControlsHidden] = useState(false);
    72→    72→  const [fs, setFs] = useState(false);
    73→    73→  const [scrub, setScrub] = useState<number | null>(null); // seek fraction while dragging
    74→    74→
    75→    75→  // Controls follow YouTube's rules: visible while the mouse is over the
    76→    76→  // video (or after a touch tap), hidden the moment it leaves, and during
    77→    77→  // playback they fade out after 3s without mouse movement. All visibility
    78→    78→  // changes are driven by events (player state, mouse) — never by effects.
    79→    79→  // Reset-style: every call clears any pending hide, so continuous mouse
    80→    80→  // movement keeps controls up and only true 3s idleness hides them.
    81→    81→  // (The old guard-against-re-schedule version hid the bar mid-hover because
    82→    82→  // the first-scheduled timeout was never extended by later movement.)
    83→    83→  const scheduleHide = (ms = 3000) => {
    84→    84→    cancelHide();
    85→    85→    idleTimer.current = setTimeout(() => {
    86→    86→      idleTimer.current = null;
    87→    87→      setControlsHidden(true);
    88→    88→    }, ms);
    89→    89→  };
    90→    90→
    91→    91→  const cancelHide = () => {
    92→    92→    if (idleTimer.current) {
    93→    93→      clearTimeout(idleTimer.current);
    94→    94→      idleTimer.current = null;
    95→    95→    }
    96→    96→  };
    97→    97→
    98→    98→  // Auto-advance bookkeeping for the ended overlay. Cancelled whenever the
    99→    99→  // user navigates elsewhere, replays, or leaves the watch page.
   100→   100→  const clearEnd = () => {
   101→   101→    if (endRef.current) {
   102→   102→      clearTimeout(endRef.current);
   103→   103→      endRef.current = null;
   104→   104→    }
   105→   105→    if (endIntRef.current) {
   106→   106→      clearInterval(endIntRef.current);
   107→   107→      endIntRef.current = null;
   108→   108→    }
   109→   109→    setCountdown(null);
   110→   110→  };
   111→   111→
   112→   112→  const replay = () => {
   113→   113→    clearEnd();
   114→   114→    const p = playerRef.current;
   115→   115→    try {
   116→   116→      p?.seekTo(0, true);
   117→   117→      p?.playVideo();
   118→   118→    } catch {
   119→   119→      /* not ready */
   120→   120→    }
   121→   121→  };
   122→   122→
   123→   123→  const playNextNow = () => {
   124→   124→    clearEnd();
   125→   125→    if (next) onSelect(next);
   126→   126→  };
   127→   127→
   128→   128→  const togglePlay = () => {
   129→   129→    const p = playerRef.current;
   130→   130→    if (!p) return;
   131→   131→    try {
   132→   132→      if (p.getPlayerState?.() === 1) p.pauseVideo();
   133→   133→      else p.playVideo();
   134→   134→    } catch {
   135→   135→      /* not ready */
   136→   136→    }
   137→   137→  };
   138→   138→
   139→   139→  const toggleMute = () => {
   140→   140→    const p = playerRef.current;
   141→   141→    if (!p) return;
   142→   142→    try {
   143→   143→      if (p.isMuted?.()) p.unMute();
   144→   144→      else p.mute();
   145→   145→    } catch {
   146→   146→      /* not ready */
   147→   147→    }
   148→   148→  };
   149→   149→
   150→   150→  const changeVol = (v: number) => {
   151→   151→    const p = playerRef.current;
   152→   152→    if (!p) return;
   153→   153→    try {
   154→   154→      p.setVolume(v);
   155→   155→      if (v > 0 && p.isMuted?.()) p.unMute();
   156→   156→    } catch {
   157→   157→      /* not ready */
   158→   158→    }
   159→   159→    setVol(v);
   160→   160→  };
   161→   161→
   162→   162→  const toggleFs = () => {
   163→   163→    const el = shellRef.current;
   164→   164→    if (!el) return;
   165→   165→    if (document.fullscreenElement) document.exitFullscreen().catch(() => {});
   166→   166→    else el.requestFullscreen?.().catch(() => {});
   167→   167→  };
   168→   168→
   169→   169→  const seekTo = (frac: number) => {
   170→   170→    const p = playerRef.current;
   171→   171→    if (!p || !dur) return;
   172→   172→    try {
   173→   173→      p.seekTo(frac * dur, true);
   174→   174→    } catch {
   175→   175→      /* not ready */
   176→   176→    }
   177→   177→  };
   178→   178→
   179→   179→  // Click on the video toggles playback; double-click goes fullscreen.
   180→   180→  // The pending click is cancelled when a double-click lands.
   181→   181→  const onShellClick = () => {
   182→   182→    if (touchRef.current) {
   183→   183→      // Mobile-style: tap toggles the chrome; auto-hide only while playing.
   184→   184→      if (controlsHidden) {
   185→   185→        setControlsHidden(false);
   186→   186→        setTouchUi(true);
   187→   187→        if (pstateRef.current === 1) scheduleHide();
   188→   188→      } else {
   189→   189→        cancelHide();
   190→   190→        setControlsHidden(true);
   191→   191→      }
   192→   192→      return;
   193→   193→    }
   194→   194→    if (clickTimer.current) return;
   195→   195→    clickTimer.current = setTimeout(() => {
   196→   196→      clickTimer.current = null;
   197→   197→      togglePlay();
   198→   198→    }, 250);
   199→   199→  };
   200→   200→
   201→   201→  const onShellDoubleClick = () => {
   202→   202→    if (clickTimer.current) {
   203→   203→      clearTimeout(clickTimer.current);
   204→   204→      clickTimer.current = null;
   205→   205→    }
   206→   206→    toggleFs();
   207→   207→  };
   208→   208→
   209→   209→  const onOverlayMouseMove = (e: React.MouseEvent) => {
   210→   210→    setHover(true);
   211→   211→    const inBar = !!(e.target as HTMLElement | null)?.closest?.("[data-ct-bar]");
   212→   212→    if (inBar) {
   213→   213→      // Interacting with the bar itself: pin controls, never fade mid-drag.
   214→   214→      cancelHide();
   215→   215→      setControlsHidden(false);
   216→   216→    } else if (pstateRef.current === 1) {
   217→   217→      setControlsHidden(false);
   218→   218→      scheduleHide(); // reset on every move -> only true idleness hides
   219→   219→    } else {
   220→   220→      // paused / buffering / ended -> controls stay up while hovered
   221→   221→      cancelHide();
   222→   222→      setControlsHidden(false);
   223→   223→    }
   224→   224→  };
   225→   225→
   226→   226→  const onShellMouseLeave = () => {
   227→   227→    setHover(false);
   228→   228→    setTouchUi(false);
   229→   229→    cancelHide();
   230→   230→    setControlsHidden(true); // mouse off the video -> always hide
   231→   231→  };
   232→   232→
   233→   233→  const { hostRef, playerRef } = useYTPlayer({
   234→   234→    videoId: video.id,
   235→   235→    autoplay: true,
   236→   236→    onState: (s) => {
   237→   237→      pstateRef.current = s;
   238→   238→      setPstate(s);
   239→   239→      if (s === 1) scheduleHide();
   240→   240→      else {
   241→   241→        cancelHide();
   242→   242→        setControlsHidden(false); // paused/buffering/ended -> show while hovered
   243→   243→      }
   244→   244→    },
   245→   245→    onEnd: () => {
   246→   246→      if (autoplay && next) {
   247→   247→        // Our own end card replaces YouTube's end screen; advance after 5s.
   248→   248→        setCountdown(5);
   249→   249→        endIntRef.current = setInterval(() => {
   250→   250→          setCountdown((c) => (c !== null && c > 1 ? c - 1 : null));
   251→   251→        }, 1000);
   252→   252→        endRef.current = setTimeout(() => {
   253→   253→          clearEnd();
   254→   254→          onSelect(next);
   255→   255→        }, 5000);
   256→   256→      }
   257→   257→    },
   258→   258→  });
   259→   259→
   260→   260→  useEffect(
   261→   261→    () => () => {
   262→   262→      if (endRef.current) clearTimeout(endRef.current);
   263→   263→      if (endIntRef.current) clearInterval(endIntRef.current);
   264→   264→    },
   265→   265→    []
   266→   266→  );
   267→   267→
   268→   268→  // A pending auto-advance belongs to the video that just ended — navigating
   269→   269→  // to any other video must cancel it instead of hijacking the navigation.
   270→   270→  useEffect(() => {
   271→   271→    clearEnd();
   272→   272→    // eslint-disable-next-line react-hooks/exhaustive-deps
   273→   273→  }, [video.id]);
   274→   274→
   275→   275→  // Player-state polling + controls visibility machinery.
   276→   276→  useEffect(() => {
   277→   277→    try {
   278→   278→      touchRef.current = window.matchMedia?.("(pointer: coarse)").matches ?? false;
   279→   279→    } catch {
   280→   280→      /* noop */
   281→   281→    }
   282→   282→    const poll = setInterval(() => {
   283→   283→      const p = playerRef.current;
   284→   284→      if (!p || typeof p.getCurrentTime !== "function") return;
   285→   285→      try {
   286→   286→        setCur(p.getCurrentTime() || 0);
   287→   287→        setDur(p.getDuration() || 0);
   288→   288→        setBuf(p.getVideoLoadedFraction() || 0);
   289→   289→        setMuted(!!p.isMuted?.());
   290→   290→        const v = p.getVolume?.();
   291→   291→        if (typeof v === "number") setVol(v);
   292→   292→      } catch {
   293→   293→        /* player mid-swap */
   294→   294→      }
   295→   295→    }, 250);
   296→   296→    const onFs = () => setFs(document.fullscreenElement === shellRef.current);
   297→   297→    document.addEventListener("fullscreenchange", onFs);
   298→   298→    return () => {
   299→   299→      clearInterval(poll);
   300→   300→      document.removeEventListener("fullscreenchange", onFs);
   301→   301→      if (idleTimer.current) clearTimeout(idleTimer.current);
   302→   302→      if (clickTimer.current) clearTimeout(clickTimer.current);
   303→   303→    };
   304→   304→  }, []);
   305→   305→
   306→   306→  // YouTube-style shortcuts (t = theater, f = fullscreen, m = mute,
   307→   307→  // space/k = play-pause, arrows = seek). Inline player calls keep this
   308→   308→  // effect's dependencies to just the theater props.
   309→   309→  useEffect(() => {
   310→   310→    const onKey = (e: KeyboardEvent) => {
   311→   311→      if (e.metaKey || e.ctrlKey || e.altKey) return;
   312→   312→      const el = e.target as HTMLElement | null;
   313→   313→      if (
   314→   314→        el &&
   315→   315→        (el.tagName === "INPUT" ||
   316→   316→          el.tagName === "TEXTAREA" ||
   317→   317→          el.isContentEditable)
   318→   318→      )
   319→   319→        return;
   320→   320→      if (e.key === "t") {
   321→   321→        onTheaterChange(!theater);
   322→   322→        return;
   323→   323→      }
   324→   324→      const p = playerRef.current;
   325→   325→      if (!p) return;
   326→   326→      try {
   327→   327→        if (e.key === " " || e.key === "k") {
   328→   328→          e.preventDefault();
   329→   329→          if (p.getPlayerState?.() === 1) p.pauseVideo();
   330→   330→          else p.playVideo();
   331→   331→        } else if (e.key === "ArrowRight") {
   332→   332→          e.preventDefault();
   333→   333→          p.seekTo(p.getCurrentTime() + 5, true);
   334→   334→        } else if (e.key === "ArrowLeft") {
   335→   335→          e.preventDefault();
   336→   336→          p.seekTo(Math.max(0, p.getCurrentTime() - 5), true);
   337→   337→        } else if (e.key === "m") {
   338→   338→          if (p.isMuted?.()) p.unMute();
   339→   339→          else p.mute();
   340→   340→        } else if (e.key === "f") {
   341→   341→          toggleFs();
   342→   342→        }
   343→   343→      } catch {
   344→   344→        /* not ready */
   345→   345→      }
   346→   346→    };
   347→   347→    window.addEventListener("keydown", onKey);
   348→   348→    return () => window.removeEventListener("keydown", onKey);
   349→   349→  }, [theater, onTheaterChange, playerRef]);
   350→   350→
   351→   351→  return (
   352→   352→    <>
   353→   353→      <div className="mx-auto flex max-w-[1700px] flex-wrap items-center gap-3 px-4 pb-3 pt-4">
   354→   354→        <Button variant="ghost" size="sm" onClick={onBack} className="gap-2">
   355→   355→          <ArrowLeft className="h-4 w-4" /> Back
   356→   356→        </Button>
   357→   357→        {idx >= 0 && (
   358→   358→          <span className="text-sm text-muted-foreground">
   359→   359→            {idx + 1} / {queue.length} in queue
   360→   360→          </span>
   361→   361→        )}
   362→   362→        <div className="ml-auto flex items-center gap-3">
   363→   363→          <label className="flex items-center gap-2 text-sm text-muted-foreground">
   364→   364→            <Switch
   365→   365→              checked={autoplay}
   366→   366→              onCheckedChange={onAutoplayChange}
   367→   367→              aria-label="Autoplay next"
   368→   368→            />
   369→   369→            Autoplay
   370→   370→          </label>
   371→   371→          <Button
   372→   372→            variant="outline"
   373→   373→            size="sm"
   374→   374→            onClick={() => onTheaterChange(!theater)}
   375→   375→            aria-pressed={theater}
   376→   376→            title="Theater mode (t)"
   377→   377→            className="gap-2"
   378→   378→          >
   379→   379→            <RectangleHorizontal className="h-4 w-4" /> Theater
   380→   380→          </Button>
   381→   381→          {next && (
   382→   382→            <Button
   383→   383→              size="sm"
   384→   384→              onClick={() => onSelect(next)}
   385→   385→              className="gap-2 bg-[#e5484d] text-white hover:bg-[#d23c41]"
   386→   386→            >
   387→   387→              Next <SkipForward className="h-4 w-4" />
   388→   388→            </Button>
   389→   389→          )}
   390→   390→        </div>
   391→   391→      </div>
   392→   392→
   393→   393→      <div className={theater ? "pb-6" : "mx-auto max-w-[1700px] px-4 pb-4"}>
   394→   394→        {/*
   395→   395→          Theater collapses the two-column grid to a plain block so the player
   396→   396→          bleeds edge-to-edge. The player host div keeps the exact same tree
   397→   397→          position in both modes — swapping only its classes/size — so the
   398→   398→          YouTube iframe is never unmounted and playback continues seamlessly.
   399→   399→        */}
   400→   400→        <div
   401→   401→          className={
   402→   402→            theater
   403→   403→              ? ""
   404→   404→              : "grid gap-6 lg:grid-cols-[minmax(0,1fr)_360px] xl:grid-cols-[minmax(0,1fr)_400px]"
   405→   405→          }
   406→   406→        >
   407→   407→          <div className="min-w-0">
   408→   408→            {(() => {
   409→   409→              const show = (hover || touchUi) && !controlsHidden;
   410→   410→              const playedPct = dur ? (scrub ?? cur / dur) * 100 : 0;
   411→   411→              const bufPct = Math.max(playedPct, buf * 100);
   412→   412→              const seekGrad = `linear-gradient(to right, #e5484d 0%, #e5484d ${playedPct}%, rgba(255,255,255,.55) ${playedPct}%, rgba(255,255,255,.55) ${bufPct}%, rgba(255,255,255,.25) ${bufPct}%)`;
   413→   413→              const volPct = muted ? 0 : vol;
   414→   414→              const volGrad = `linear-gradient(to right, #fff 0%, #fff ${volPct}%, rgba(255,255,255,.25) ${volPct}%)`;
   415→   415→              return (
   416→   416→                <div
   417→   417→                  ref={shellRef}
   418→   418→                  className={cn(
   419→   419→                    "relative overflow-hidden bg-black",
   420→   420→                    theater ? "w-full" : "aspect-video w-full rounded-xl"
   421→   421→                  )}
   422→   422→                  style={theater ? { height: "min(85vh, 56.25vw)" } : undefined}
   423→   423→                >
   424→   424→                  <div ref={hostRef} className="h-full w-full" />
   425→   425→
   426→   426→                  {/*
   427→   427→                    Event-capture overlay: the cross-origin YouTube iframe is
   428→   428→                    out-of-process and never reports mouse movement to this
   429→   429→                    page, so a transparent layer above it drives hover
   430→   430→                    detection, click-to-pause and double-click fullscreen.
   431→   431→                    Playback itself is driven through the IFrame API.
   432→   432→                  */}
   433→   433→                  <div
   434→   434→                    className={cn("absolute inset-0", !show && "cursor-none")}
   435→   435→                    onMouseEnter={() => {
   436→   436→                      setHover(true);
   437→   437→                      setControlsHidden(false); // show instantly, no wiggle needed
   438→   438→                      if (pstateRef.current === 1) scheduleHide();
   439→   439→                    }}
   440→   440→                    onMouseMove={onOverlayMouseMove}
   441→   441→                    onMouseLeave={onShellMouseLeave}
   442→   442→                    onClick={onShellClick}
   443→   443→                    onDoubleClick={onShellDoubleClick}
   444→   444→                  >
   445→   445→                    {/* Hover-only control bar — YouTube-style. The bar lives
   446→   446→                        inside the overlay so moving onto it keeps controls up. */}
   447→   447→                    <div
   448→   448→                      data-ct-bar
   449→   449→                      onClick={(e) => e.stopPropagation()}
   450→   450→                      onDoubleClick={(e) => e.stopPropagation()}
   451→   451→                      className={cn(
   452→   452→                        "absolute inset-x-0 bottom-0 z-10 bg-gradient-to-t from-black/85 via-black/40 to-transparent pb-1 pt-8 transition-opacity duration-200",
   453→   453→                        show ? "opacity-100" : "pointer-events-none opacity-0"
   454→   454→                      )}
   455→   455→                    >
   456→   456→                    <div className="px-3">
   457→   457→                      <input
   458→   458→                        type="range"
   459→   459→                        min={0}
   460→   460→                        max={1000}
   461→   461→                        step={1}
   462→   462→                        aria-label="Seek"
   463→   463→                        className="ct-range h-1 w-full"
   464→   464→                        style={{ background: seekGrad }}
   465→   465→                        value={Math.round((scrub ?? (dur ? cur / dur : 0)) * 1000)}
   466→   466→                        onChange={(e) => {
   467→   467→                          const frac = Number(e.target.value) / 1000;
   468→   468→                          setScrub(frac);
   469→   469→                          seekTo(frac);
   470→   470→                        }}
   471→   471→                        onPointerDown={() => {
   472→   472→                          if (touchRef.current) setTouchUi(true);
   473→   473→                        }}
   474→   474→                        onPointerUp={() => setScrub(null)}
   475→   475→                        onKeyUp={() => setScrub(null)}
   476→   476→                        onBlur={() => setScrub(null)}
   477→   477→                      />
