     2→     2→

     4→     4→import {
     5→     5→  ArrowLeft,
     6→     6→  Maximize,
     7→     7→  Minimize,

    26→    26→interface WatchViewProps {
    27→    27→  video: Vid;
    28→    28→  queue: Vid[];
    29→    29→  channels: Chan[];
    30→    30→  autoplay: boolean;
    31→    31→  theater: boolean;
    32→    32→  onAutoplayChange: (v: boolean) => void;
    33→    33→  onTheaterChange: (v: boolean) => void;
    34→    34→  onBack: () => void;
    35→    35→  onSelect: (v: Vid) => void;

    38→    38→
    39→    39→export function WatchView({
    40→    40→  video,
    41→    41→  queue,
    42→    42→  channels,
    43→    43→  autoplay,
    44→    44→  theater,
    45→    45→  onAutoplayChange,
    46→    46→  onTheaterChange,
    47→    47→  onBack,
    48→    48→  onSelect,
    49→    49→  onTv,
    50→    50→}: WatchViewProps) {
    51→    51→  const idx = queue.findIndex((v) => v.id === video.id);
    52→    52→  const next =
    53→    53→    idx >= 0 && idx + 1 < queue.length ? queue[idx + 1] : (queue[0] ?? null);
    54→    54→  const chan = channels[video.c];
    55→    55→  const upNext = idx >= 0 ? queue.slice(idx + 1, idx + 26) : queue.slice(0, 25);
    56→    56→  const endRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    57→    57→  const endIntRef = useRef<ReturnType<typeof setInterval> | null>(null);
    58→    58→  const [countdown, setCountdown] = useState<number | null>(null);
    59→    59→  // True while we are leaving an ENDED video and the next one is still
    60→    60→  // buffering. An opaque cover stays up for this window — otherwise the
    61→    61→  // previous video's YouTube end screen (share / watch-later / "More
    62→    62→  // videos" / logo) leaks through for a few seconds at the start.
    63→    63→  const [swapping, setSwapping] = useState(false);
    64→    64→
    65→    65→  // ----- custom player chrome (hover-only controls) -----
    66→    66→  const shellRef = useRef<HTMLDivElement | null>(null);
    67→    67→  const idleTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    68→    68→  const clickTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    69→    69→  const touchRef = useRef(false);
    70→    70→  const pstateRef = useRef(-1);
    71→    71→  const [pstate, setPstate] = useState(-1); // YT player state (-1 unstarted .. 3 buffering)
    72→    72→  const [cur, setCur] = useState(0);
    73→    73→  const [dur, setDur] = useState(0);
    74→    74→  const [buf, setBuf] = useState(0);
    75→    75→  const [muted, setMuted] = useState(false);
    76→    76→  const [vol, setVol] = useState(100);
    77→    77→  const [hover, setHover] = useState(false);
    78→    78→  const [touchUi, setTouchUi] = useState(false);
    79→    79→  const [controlsHidden, setControlsHidden] = useState(false);
    80→    80→  const [fs, setFs] = useState(false);
    81→    81→  const [scrub, setScrub] = useState<number | null>(null); // seek fraction while dragging
    82→    82→
    83→    83→  // Controls follow YouTube's rules: visible while the mouse is over the
    84→    84→  // video (or after a touch tap), hidden the moment it leaves, and during
    85→    85→  // playback they fade out after 3s without mouse movement. All visibility
    86→    86→  // changes are driven by events (player state, mouse) — never by effects.
    87→    87→  // Reset-style: every call clears any pending hide, so continuous mouse
    88→    88→  // movement keeps controls up and only true 3s idleness hides them.
    89→    89→  // (The old guard-against-re-schedule version hid the bar mid-hover because
    90→    90→  // the first-scheduled timeout was never extended by later movement.)
    91→    91→  const scheduleHide = (ms = 3000) => {
    92→    92→    cancelHide();
    93→    93→    idleTimer.current = setTimeout(() => {
    94→    94→      idleTimer.current = null;
    95→    95→      setControlsHidden(true);
    96→    96→    }, ms);
    97→    97→  };
    98→    98→
    99→    99→  const cancelHide = () => {
   100→   100→    if (idleTimer.current) {
   101→   101→      clearTimeout(idleTimer.current);
   102→   102→      idleTimer.current = null;
   103→   103→    }
   104→   104→  };
   105→   105→
   106→   106→  // Auto-advance bookkeeping for the ended overlay. Cancelled whenever the
   107→   107→  // user navigates elsewhere, replays, or leaves the watch page.
   108→   108→  const clearEnd = () => {
   109→   109→    if (endRef.current) {
   110→   110→      clearTimeout(endRef.current);
   111→   111→      endRef.current = null;
   112→   112→    }
   113→   113→    if (endIntRef.current) {
   114→   114→      clearInterval(endIntRef.current);
   115→   115→      endIntRef.current = null;
   116→   116→    }
   117→   117→    setCountdown(null);
   118→   118→  };
   119→   119→
   120→   120→  const replay = () => {
   121→   121→    clearEnd();
   122→   122→    setSwapping(true); // cover the old end screen while it re-buffers
   123→   123→    const p = playerRef.current;
   124→   124→    try {
   125→   125→      p?.seekTo(0, true);
   126→   126→      p?.playVideo();
   127→   127→    } catch {
   128→   128→      /* not ready */
   129→   129→    }
   130→   130→  };
   131→   131→
   132→   132→  // Navigate to another video. When the current one has ENDED, raise the
   133→   133→  // swap cover: loadVideoById keeps the old iframe content (end screen
   134→   134→  // included) visible until the new video's first frame renders.
   135→   135→  const selectVideo = (v: Vid) => {
   136→   136→    if (pstateRef.current === 0) setSwapping(true);
   137→   137→    clearEnd();
   138→   138→    onSelect(v);
   139→   139→  };
   140→   140→
   141→   141→  const togglePlay = () => {
   142→   142→    const p = playerRef.current;
   143→   143→    if (!p) return;
   144→   144→    try {
   145→   145→      if (p.getPlayerState?.() === 1) p.pauseVideo();
   146→   146→      else p.playVideo();
   147→   147→    } catch {
   148→   148→      /* not ready */
   149→   149→    }
   150→   150→  };
   151→   151→
   152→   152→  const toggleMute = () => {
   153→   153→    const p = playerRef.current;
   154→   154→    if (!p) return;
   155→   155→    try {
   156→   156→      if (p.isMuted?.()) p.unMute();
   157→   157→      else p.mute();
   158→   158→    } catch {
   159→   159→      /* not ready */
   160→   160→    }
   161→   161→  };
   162→   162→
   163→   163→  const changeVol = (v: number) => {
   164→   164→    const p = playerRef.current;
   165→   165→    if (!p) return;
   166→   166→    try {
   167→   167→      p.setVolume(v);
   168→   168→      if (v > 0 && p.isMuted?.()) p.unMute();
   169→   169→    } catch {
   170→   170→      /* not ready */
   171→   171→    }
   172→   172→    setVol(v);
   173→   173→  };
   174→   174→
   175→   175→  const toggleFs = () => {
   176→   176→    const el = shellRef.current;
   177→   177→    if (!el) return;
   178→   178→    if (document.fullscreenElement) document.exitFullscreen().catch(() => {});
   179→   179→    else el.requestFullscreen?.().catch(() => {});
   180→   180→  };
   181→   181→
   182→   182→  const seekTo = (frac: number) => {
   183→   183→    const p = playerRef.current;
   184→   184→    if (!p || !dur) return;
   185→   185→    try {
   186→   186→      p.seekTo(frac * dur, true);
   187→   187→    } catch {
   188→   188→      /* not ready */
   189→   189→    }
   190→   190→  };
   191→   191→
   192→   192→  // Click on the video toggles playback; double-click goes fullscreen.
   193→   193→  // The pending click is cancelled when a double-click lands.
   194→   194→  const onShellClick = () => {
   195→   195→    if (touchRef.current) {
   196→   196→      // Mobile-style: tap toggles the chrome; auto-hide only while playing.
   197→   197→      if (controlsHidden) {
   198→   198→        setControlsHidden(false);
   199→   199→        setTouchUi(true);
   200→   200→        if (pstateRef.current === 1) scheduleHide();
   201→   201→      } else {
   202→   202→        cancelHide();
   203→   203→        setControlsHidden(true);
   204→   204→      }
   205→   205→      return;
   206→   206→    }
   207→   207→    if (clickTimer.current) return;
   208→   208→    clickTimer.current = setTimeout(() => {
   209→   209→      clickTimer.current = null;
   210→   210→      togglePlay();
   211→   211→    }, 250);
   212→   212→  };
   213→   213→
   214→   214→  const onShellDoubleClick = () => {
   215→   215→    if (clickTimer.current) {
   216→   216→      clearTimeout(clickTimer.current);
   217→   217→      clickTimer.current = null;
   218→   218→    }
   219→   219→    toggleFs();
   220→   220→  };
   221→   221→
   222→   222→  const onOverlayMouseMove = (e: React.MouseEvent) => {
   223→   223→    setHover(true);
   224→   224→    const inBar = !!(e.target as HTMLElement | null)?.closest?.("[data-ct-bar]");
   225→   225→    if (inBar) {
   226→   226→      // Interacting with the bar itself: pin controls, never fade mid-drag.
   227→   227→      cancelHide();
   228→   228→      setControlsHidden(false);
   229→   229→    } else if (pstateRef.current === 1) {
   230→   230→      setControlsHidden(false);
   231→   231→      scheduleHide(); // reset on every move -> only true idleness hides
   232→   232→    } else {
   233→   233→      // paused / buffering / ended -> controls stay up while hovered
   234→   234→      cancelHide();
   235→   235→      setControlsHidden(false);
   236→   236→    }
   237→   237→  };
   238→   238→
   239→   239→  const onShellMouseLeave = () => {
   240→   240→    setHover(false);
   241→   241→    setTouchUi(false);
   242→   242→    cancelHide();
   243→   243→    setControlsHidden(true); // mouse off the video -> always hide
   244→   244→  };
   245→   245→
   246→   246→  const { hostRef, playerRef } = useYTPlayer({
   247→   247→    videoId: video.id,
   248→   248→    autoplay: true,
   249→   249→    onState: (s) => {
   250→   250→      pstateRef.current = s;
   251→   251→      setPstate(s);
   252→   252→      if (s === 1 || s === 2) setSwapping(false); // new video is on screen
   253→   253→      if (s === 1) scheduleHide();
   254→   254→      else {
   255→   255→        cancelHide();
   256→   256→        setControlsHidden(false); // paused/buffering/ended -> show while hovered
   257→   257→      }
   258→   258→    },
   259→   259→    onEnd: () => {
   260→   260→      if (autoplay && next) {
   261→   261→        // Our own end card replaces YouTube's end screen; advance after 5s.
   262→   262→        setCountdown(5);
   263→   263→        endIntRef.current = setInterval(() => {
   264→   264→          setCountdown((c) => (c !== null && c > 1 ? c - 1 : null));
   265→   265→        }, 1000);
   266→   266→        endRef.current = setTimeout(() => {
   267→   267→          clearEnd();
   268→   268→          selectVideo(next);
   269→   269→        }, 5000);
   270→   270→      }
   271→   271→    },
   272→   272→  });
   273→   273→
   274→   274→  useEffect(
   275→   275→    () => () => {
   276→   276→      if (endRef.current) clearTimeout(endRef.current);
   277→   277→      if (endIntRef.current) clearInterval(endIntRef.current);
   278→   278→    },
   279→   279→    []
   280→   280→  );
   281→   281→
   282→   282→  // A pending auto-advance belongs to the video that just ended — navigating
   283→   283→  // to any other video must cancel it instead of hijacking the navigation.
   284→   284→  useEffect(() => {
   285→   285→    clearEnd();
   286→   286→    // eslint-disable-next-line react-hooks/exhaustive-deps
   287→   287→  }, [video.id]);
   288→   288→
   289→   289→  // Player-state polling + controls visibility machinery.
   290→   290→  useEffect(() => {
   291→   291→    try {
   292→   292→      touchRef.current = window.matchMedia?.("(pointer: coarse)").matches ?? false;
   293→   293→    } catch {
   294→   294→      /* noop */
   295→   295→    }
   296→   296→    const poll = setInterval(() => {
   297→   297→      const p = playerRef.current;
   298→   298→      if (!p || typeof p.getCurrentTime !== "function") return;
   299→   299→      try {
   300→   300→        setCur(p.getCurrentTime() || 0);
   301→   301→        setDur(p.getDuration() || 0);
   302→   302→        setBuf(p.getVideoLoadedFraction() || 0);
   303→   303→        setMuted(!!p.isMuted?.());
   304→   304→        const v = p.getVolume?.();
   305→   305→        if (typeof v === "number") setVol(v);
   306→   306→      } catch {
   307→   307→        /* player mid-swap */
   308→   308→      }
   309→   309→    }, 250);
   310→   310→    const onFs = () => setFs(document.fullscreenElement === shellRef.current);
   311→   311→    document.addEventListener("fullscreenchange", onFs);
   312→   312→    return () => {
   313→   313→      clearInterval(poll);
   314→   314→      document.removeEventListener("fullscreenchange", onFs);
   315→   315→      if (idleTimer.current) clearTimeout(idleTimer.current);
   316→   316→      if (clickTimer.current) clearTimeout(clickTimer.current);
   317→   317→    };
   318→   318→  }, []);
   319→   319→
   320→   320→  // YouTube-style shortcuts (t = theater, f = fullscreen, m = mute,
   321→   321→  // space/k = play-pause, arrows = seek). Inline player calls keep this
   322→   322→  // effect's dependencies to just the theater props.
   323→   323→  useEffect(() => {
   324→   324→    const onKey = (e: KeyboardEvent) => {
   325→   325→      if (e.metaKey || e.ctrlKey || e.altKey) return;
   326→   326→      const el = e.target as HTMLElement | null;
   327→   327→      if (
   328→   328→        el &&
   329→   329→        (el.tagName === "INPUT" ||
   330→   330→          el.tagName === "TEXTAREA" ||
   331→   331→          el.isContentEditable)
   332→   332→      )
   333→   333→        return;
   334→   334→      if (e.key === "t") {
   335→   335→        onTheaterChange(!theater);
   336→   336→        return;
   337→   337→      }
   338→   338→      const p = playerRef.current;
   339→   339→      if (!p) return;
   340→   340→      try {
   341→   341→        if (e.key === " " || e.key === "k") {
   342→   342→          e.preventDefault();
   343→   343→          if (p.getPlayerState?.() === 1) p.pauseVideo();
   344→   344→          else p.playVideo();
   345→   345→        } else if (e.key === "ArrowRight") {
   346→   346→          e.preventDefault();
   347→   347→          p.seekTo(p.getCurrentTime() + 5, true);
   348→   348→        } else if (e.key === "ArrowLeft") {
   349→   349→          e.preventDefault();
   350→   350→          p.seekTo(Math.max(0, p.getCurrentTime() - 5), true);
   351→   351→        } else if (e.key === "m") {
   352→   352→          if (p.isMuted?.()) p.unMute();
   353→   353→          else p.mute();
   354→   354→        } else if (e.key === "f") {
   355→   355→          toggleFs();
   356→   356→        }
   357→   357→      } catch {
   358→   358→        /* not ready */
   359→   359→      }
   360→   360→    };
   361→   361→    window.addEventListener("keydown", onKey);
   362→   362→    return () => window.removeEventListener("keydown", onKey);
   363→   363→  }, [theater, onTheaterChange, playerRef]);
   364→   364→
   365→   365→  return (
   366→   366→    <>
   367→   367→      <div className="mx-auto flex max-w-[1700px] flex-wrap items-center gap-3 px-4 pb-3 pt-4">
   368→   368→        <Button variant="ghost" size="sm" onClick={onBack} className="gap-2">
   369→   369→          <ArrowLeft className="h-4 w-4" /> Back
   370→   370→        </Button>
   371→   371→        {idx >= 0 && (
   372→   372→          <span className="text-sm text-muted-foreground">
   373→   373→            {idx + 1} / {queue.length} in queue
   374→   374→          </span>
   375→   375→        )}
   376→   376→        <div className="ml-auto flex items-center gap-3">
   377→   377→          <label className="flex items-center gap-2 text-sm text-muted-foreground">
   378→   378→            <Switch
   379→   379→              checked={autoplay}
   380→   380→              onCheckedChange={onAutoplayChange}
   381→   381→              aria-label="Autoplay next"
   382→   382→            />
   383→   383→            Autoplay
   384→   384→          </label>
   385→   385→          <Button
   386→   386→            variant="outline"
   387→   387→            size="sm"
   388→   388→            onClick={() => onTheaterChange(!theater)}
   389→   389→            aria-pressed={theater}
   390→   390→            title="Theater mode (t)"
   391→   391→            className="gap-2"
   392→   392→          >
   393→   393→            <RectangleHorizontal className="h-4 w-4" /> Theater
   394→   394→          </Button>
   395→   395→          <Button variant="outline" size="sm" onClick={onTv} className="gap-2">
   396→   396→            <Tv className="h-4 w-4" /> TV mode
   397→   397→          </Button>
   398→   398→          {next && (
   399→   399→            <Button
   400→   400→              size="sm"
   401→   401→              onClick={() => selectVideo(next)}
   402→   402→              className="gap-2 bg-[#e5484d] text-white hover:bg-[#d23c41]"
   403→   403→            >
   404→   404→              Next <SkipForward className="h-4 w-4" />
   405→   405→            </Button>
   406→   406→          )}
   407→   407→        </div>
   408→   408→      </div>
   409→   409→
   410→   410→      <div className={theater ? "pb-6" : "mx-auto max-w-[1700px] px-4 pb-4"}>
   411→   411→        {/*
   412→   412→          Theater collapses the two-column grid to a plain block so the player
   413→   413→          bleeds edge-to-edge. The player host div keeps the exact same tree
   414→   414→          position in both modes — swapping only its classes/size — so the
   415→   415→          YouTube iframe is never unmounted and playback continues seamlessly.
   416→   416→        */}
   417→   417→        <div
   418→   418→          className={
   419→   419→            theater
   420→   420→              ? ""
   421→   421→              : "grid gap-6 lg:grid-cols-[minmax(0,1fr)_360px] xl:grid-cols-[minmax(0,1fr)_400px]"
   422→   422→          }
   423→   423→        >
   424→   424→          <div className="min-w-0">
   425→   425→            {(() => {
   426→   426→              const show = (hover || touchUi) && !controlsHidden;
   427→   427→              const playedPct = dur ? (scrub ?? cur / dur) * 100 : 0;
   428→   428→              const bufPct = Math.max(playedPct, buf * 100);
   429→   429→              const seekGrad = `linear-gradient(to right, #e5484d 0%, #e5484d ${playedPct}%, rgba(255,255,255,.55) ${playedPct}%, rgba(255,255,255,.55) ${bufPct}%, rgba(255,255,255,.25) ${bufPct}%)`;
   430→   430→              const volPct = muted ? 0 : vol;
   431→   431→              const volGrad = `linear-gradient(to right, #fff 0%, #fff ${volPct}%, rgba(255,255,255,.25) ${volPct}%)`;
   432→   432→              return (
   433→   433→                <div
   434→   434→                  ref={shellRef}
   435→   435→                  className={cn(
   436→   436→                    "relative overflow-hidden bg-black",
   437→   437→                    theater ? "w-full" : "aspect-video w-full rounded-xl"
   438→   438→                  )}
   439→   439→                  style={theater ? { height: "min(85vh, 56.25vw)" } : undefined}
   440→   440→                >
   441→   441→                  <div ref={hostRef} className="h-full w-full" />
   442→   442→
   443→   443→                  {/*
   444→   444→                    Event-capture overlay: the cross-origin YouTube iframe is
   445→   445→                    out-of-process and never reports mouse movement to this
   446→   446→                    page, so a transparent layer above it drives hover
   447→   447→                    detection, click-to-pause and double-click fullscreen.
   448→   448→                    Playback itself is driven through the IFrame API.
   449→   449→                  */}
   450→   450→                  <div
   451→   451→                    className={cn("absolute inset-0", !show && "cursor-none")}
   452→   452→                    onMouseEnter={() => {
   453→   453→                      setHover(true);
   454→   454→                      setControlsHidden(false); // show instantly, no wiggle needed
   455→   455→                      if (pstateRef.current === 1) scheduleHide();
   456→   456→                    }}
   457→   457→                    onMouseMove={onOverlayMouseMove}
   458→   458→                    onMouseLeave={onShellMouseLeave}
   459→   459→                    onClick={onShellClick}
   460→   460→                    onDoubleClick={onShellDoubleClick}
   461→   461→                  >
   462→   462→                    {/* Hover-only control bar — YouTube-style. The bar lives
   463→   463→                        inside the overlay so moving onto it keeps controls up. */}
   464→   464→                    <div
   465→   465→                      data-ct-bar
   466→   466→                      onClick={(e) => e.stopPropagation()}
   467→   467→                      onDoubleClick={(e) => e.stopPropagation()}
   468→   468→                      className={cn(
   469→   469→                        "absolute inset-x-0 bottom-0 z-10 bg-gradient-to-t from-black/85 via-black/40 to-transparent pb-1 pt-8 transition-opacity duration-200",
   470→   470→                        show ? "opacity-100" : "pointer-events-none opacity-0"
   471→   471→                      )}
   472→   472→                    >
