     1→"use client";
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react";
     4→import {
     5→  ArrowLeft,
     6→  Maximize,
     7→  Minimize,
     8→  Pause,
     9→  Play,
    10→  RectangleHorizontal,
    11→  SkipForward,
    12→  Tv,
    13→  Volume1,
    14→  Volume2,
    15→  VolumeX,
    16→} from "lucide-react";
    17→import { Badge } from "@/components/ui/badge";
    18→import { Button } from "@/components/ui/button";
    19→import { ScrollArea } from "@/components/ui/scroll-area";
    20→import { Switch } from "@/components/ui/switch";
    21→import { cn } from "@/lib/utils";
    22→import { useYTPlayer } from "./use-yt";
    23→import { chanName, fmtAge, fmtDur, fmtViews, metaJoin, TIER_COLORS, thumbUrl } from "./utils";
    24→import type { Chan, Tier, Vid } from "./types";
    25→
    26→interface WatchViewProps {
    27→  video: Vid;
    28→  queue: Vid[];
    29→  channels: Chan[];
    30→  autoplay: boolean;
    31→  theater: boolean;
    32→  onAutoplayChange: (v: boolean) => void;
    33→  onTheaterChange: (v: boolean) => void;
    34→  onBack: () => void;
    35→  onSelect: (v: Vid) => void;
    36→  onTv: () => void;
    37→}
    38→
    39→export function WatchView({
    40→  video,
    41→  queue,
    42→  channels,
    43→  autoplay,
    44→  theater,
    45→  onAutoplayChange,
    46→  onTheaterChange,
    47→  onBack,
    48→  onSelect,
    49→  onTv,
    50→}: WatchViewProps) {
    51→  const idx = queue.findIndex((v) => v.id === video.id);
    52→  const next =
    53→    idx >= 0 && idx + 1 < queue.length ? queue[idx + 1] : (queue[0] ?? null);
    54→  const chan = channels[video.c];
    55→  const upNext = idx >= 0 ? queue.slice(idx + 1, idx + 26) : queue.slice(0, 25);
    56→  const endRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    57→  const endIntRef = useRef<ReturnType<typeof setInterval> | null>(null);
    58→  const [countdown, setCountdown] = useState<number | null>(null);
    59→  // True while a video is loading and until its first frame actually plays.
    60→  // The YouTube iframe is REUSED across videos (loadVideoById), and it keeps
    61→  // painting the previous video's last frame — or, if that one ended, its
    62→  // full end screen (share / watch-later / "More videos" / logo) — through
    63→  // the unstarted(-1), cued(5) and buffering(3) states. An opaque cover
    64→  // hides all of that; it starts true so even the very first video opens on
    65→  // a clean loading screen instead of stale iframe content.
    66→  const [swapping, setSwapping] = useState(true);
    67→
    68→  // ----- custom player chrome (hover-only controls) -----
    69→  const shellRef = useRef<HTMLDivElement | null>(null);
    70→  const idleTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    71→  const clickTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    72→  const touchRef = useRef(false);
    73→  const pstateRef = useRef(-1);
    74→  const prevStateRef = useRef(-1); // player state before the current one
    75→  const playedRef = useRef(false); // current video has actually started playing
    76→  const [pstate, setPstate] = useState(-1); // YT player state (-1 unstarted .. 3 buffering)
    77→  const [cur, setCur] = useState(0);
    78→  const [dur, setDur] = useState(0);
    79→  const [buf, setBuf] = useState(0);
    80→  const [muted, setMuted] = useState(false);
    81→  const [vol, setVol] = useState(100);
    82→  const [hover, setHover] = useState(false);
    83→  const [touchUi, setTouchUi] = useState(false);
    84→  const [controlsHidden, setControlsHidden] = useState(false);
    85→  const [fs, setFs] = useState(false);
    86→  const [scrub, setScrub] = useState<number | null>(null); // seek fraction while dragging
    87→
    88→  // Controls follow YouTube's rules: visible while the mouse is over the
    89→  // video (or after a touch tap), hidden the moment it leaves, and during
    90→  // playback they fade out after 3s without mouse movement. All visibility
    91→  // changes are driven by events (player state, mouse) — never by effects.
    92→  // Reset-style: every call clears any pending hide, so continuous mouse
    93→  // movement keeps controls up and only true 3s idleness hides them.
    94→  // (The old guard-against-re-schedule version hid the bar mid-hover because
    95→  // the first-scheduled timeout was never extended by later movement.)
    96→  const scheduleHide = (ms = 3000) => {
    97→    cancelHide();
    98→    idleTimer.current = setTimeout(() => {
    99→      idleTimer.current = null;
   100→      setControlsHidden(true);
   101→    }, ms);
   102→  };
   103→
   104→  const cancelHide = () => {
   105→    if (idleTimer.current) {
   106→      clearTimeout(idleTimer.current);
   107→      idleTimer.current = null;
   108→    }
   109→  };
   110→
   111→  // Auto-advance bookkeeping for the ended overlay. Cancelled whenever the
   112→  // user navigates elsewhere, replays, or leaves the watch page.
   113→  const clearEnd = () => {
   114→    if (endRef.current) {
   115→      clearTimeout(endRef.current);
   116→      endRef.current = null;
   117→    }
   118→    if (endIntRef.current) {
   119→      clearInterval(endIntRef.current);
   120→      endIntRef.current = null;
   121→    }
   122→    setCountdown(null);
   123→  };
   124→
   125→  const replay = () => {
   126→    clearEnd();
   127→    setSwapping(true); // cover the old end screen while it re-buffers
   128→    const p = playerRef.current;
   129→    try {
   130→      p?.seekTo(0, true);
   131→      p?.playVideo();
   132→    } catch {
   133→      /* not ready */
   134→    }
   135→  };
   136→
   137→  // Navigate to another video. Always raise the swap cover: loadVideoById
   138→  // keeps the old iframe content — end screen if the video had ended, else
   139→  // its last frame — visible until the new video's first frame renders.
   140→  // (The [video.id] effect below raises it too; this covers the same-tick
   141→  // render so the end card never flashes on the way out.)
   142→  const selectVideo = (v: Vid) => {
   143→    setSwapping(true);
   144→    clearEnd();
   145→    onSelect(v);
   146→  };
   147→
   148→  const togglePlay = () => {
   149→    const p = playerRef.current;
   150→    if (!p) return;
   151→    try {
   152→      if (p.getPlayerState?.() === 1) p.pauseVideo();
   153→      else p.playVideo();
   154→    } catch {
   155→      /* not ready */
   156→    }
   157→  };
   158→
   159→  const toggleMute = () => {
   160→    const p = playerRef.current;
   161→    if (!p) return;
   162→    try {
   163→      if (p.isMuted?.()) p.unMute();
   164→      else p.mute();
   165→    } catch {
   166→      /* not ready */
   167→    }
   168→  };
   169→
   170→  const changeVol = (v: number) => {
   171→    const p = playerRef.current;
   172→    if (!p) return;
   173→    try {
   174→      p.setVolume(v);
   175→      if (v > 0 && p.isMuted?.()) p.unMute();
   176→    } catch {
   177→      /* not ready */
   178→    }
   179→    setVol(v);
   180→  };
   181→
   182→  const toggleFs = () => {
   183→    const el = shellRef.current;
   184→    if (!el) return;
   185→    if (document.fullscreenElement) document.exitFullscreen().catch(() => {});
   186→    else el.requestFullscreen?.().catch(() => {});
   187→  };
   188→
   189→  const seekTo = (frac: number) => {
   190→    const p = playerRef.current;
   191→    if (!p || !dur) return;
   192→    try {
   193→      p.seekTo(frac * dur, true);
   194→    } catch {
   195→      /* not ready */
   196→    }
   197→  };
   198→
   199→  // Click on the video toggles playback; double-click goes fullscreen.
   200→  // The pending click is cancelled when a double-click lands.
   201→  const onShellClick = () => {
   202→    if (touchRef.current) {
   203→      // Mobile-style: tap toggles the chrome; auto-hide only while playing.
   204→      if (controlsHidden) {
   205→        setControlsHidden(false);
   206→        setTouchUi(true);
   207→        if (pstateRef.current === 1) scheduleHide();
   208→      } else {
   209→        cancelHide();
   210→        setControlsHidden(true);
   211→      }
   212→      return;
   213→    }
   214→    if (clickTimer.current) return;
   215→    clickTimer.current = setTimeout(() => {
   216→      clickTimer.current = null;
   217→      togglePlay();
   218→    }, 250);
   219→  };
   220→
   221→  const onShellDoubleClick = () => {
   222→    if (clickTimer.current) {
   223→      clearTimeout(clickTimer.current);
   224→      clickTimer.current = null;
   225→    }
   226→    toggleFs();
   227→  };
   228→
   229→  const onOverlayMouseMove = (e: React.MouseEvent) => {
   230→    setHover(true);
   231→    const inBar = !!(e.target as HTMLElement | null)?.closest?.("[data-ct-bar]");
   232→    if (inBar) {
   233→      // Interacting with the bar itself: pin controls, never fade mid-drag.
   234→      cancelHide();
   235→      setControlsHidden(false);
   236→    } else if (pstateRef.current === 1) {
   237→      setControlsHidden(false);
   238→      scheduleHide(); // reset on every move -> only true idleness hides
   239→    } else {
   240→      // paused / buffering / ended -> controls stay up while hovered
   241→      cancelHide();
   242→      setControlsHidden(false);
   243→    }
   244→  };
   245→
   246→  const onShellMouseLeave = () => {
   247→    setHover(false);
   248→    setTouchUi(false);
   249→    cancelHide();
   250→    setControlsHidden(true); // mouse off the video -> always hide
   251→  };
   252→
   253→  const { hostRef, playerRef } = useYTPlayer({
   254→    videoId: video.id,
   255→    autoplay: true,
   256→    onState: (s) => {
   257→      prevStateRef.current = pstateRef.current;
   258→      pstateRef.current = s;
   259→      setPstate(s);
   260→      if (s === 1) {
   261→        playedRef.current = true;
   262→        setSwapping(false); // the new video's first frame is on screen
   263→        scheduleHide();
   264→      } else {
   265→        if (s === 2 && playedRef.current) setSwapping(false);
   266→        cancelHide();
   267→        setControlsHidden(false); // paused/buffering/ended -> show while hovered
   268→      }
   269→    },
   270→    onEnd: () => {
   271→      if (autoplay && next) {
   272→        // Our own end card replaces YouTube's end screen; advance after 5s.
   273→        setCountdown(5);
   274→        endIntRef.current = setInterval(() => {
   275→          setCountdown((c) => (c !== null && c > 1 ? c - 1 : null));
   276→        }, 1000);
   277→        endRef.current = setTimeout(() => {
   278→          clearEnd();
   279→          selectVideo(next);
   280→        }, 5000);
   281→      }
   282→    },
   283→  });
   284→
   285→  useEffect(
   286→    () => () => {
   287→      if (endRef.current) clearTimeout(endRef.current);
   288→      if (endIntRef.current) clearInterval(endIntRef.current);
   289→    },
   290→    []
   291→  );
   292→
   293→  // A pending auto-advance belongs to the video that just ended — navigating
   294→  // to any other video must cancel it instead of hijacking the navigation.
   295→  // Every video change also re-raises the swap cover: the reused iframe keeps
   296→  // showing the old video's end screen / last frame through the -1/5/3 load
   297→  // states until the new one actually plays (onState clears it then; a user
   298→  // pause of an already-started video clears it too).
   299→  useEffect(() => {
   300→    clearEnd();
   301→    playedRef.current = false;
   302→    setSwapping(true);
   303→    // eslint-disable-next-line react-hooks/exhaustive-deps
   304→  }, [video.id]);
   305→
   306→  // Player-state polling + controls visibility machinery.
   307→  useEffect(() => {
   308→    try {
   309→      touchRef.current = window.matchMedia?.("(pointer: coarse)").matches ?? false;
   310→    } catch {
   311→      /* noop */
   312→    }
   313→    const poll = setInterval(() => {
   314→      const p = playerRef.current;
   315→      if (!p || typeof p.getCurrentTime !== "function") return;
   316→      try {
   317→        setCur(p.getCurrentTime() || 0);
   318→        setDur(p.getDuration() || 0);
   319→        setBuf(p.getVideoLoadedFraction() || 0);
   320→        setMuted(!!p.isMuted?.());
   321→        const v = p.getVolume?.();
   322→        if (typeof v === "number") setVol(v);
   323→      } catch {
   324→        /* player mid-swap */
   325→      }
   326→    }, 250);
   327→    const onFs = () => setFs(document.fullscreenElement === shellRef.current);
   328→    document.addEventListener("fullscreenchange", onFs);
   329→    return () => {
   330→      clearInterval(poll);
   331→      document.removeEventListener("fullscreenchange", onFs);
   332→      if (idleTimer.current) clearTimeout(idleTimer.current);
   333→      if (clickTimer.current) clearTimeout(clickTimer.current);
   334→    };
   335→  }, []);
   336→
   337→  // YouTube-style shortcuts (t = theater, f = fullscreen, m = mute,
   338→  // space/k = play-pause, arrows = seek). Inline player calls keep this
   339→  // effect's dependencies to just the theater props.
   340→  useEffect(() => {
   341→    const onKey = (e: KeyboardEvent) => {
   342→      if (e.metaKey || e.ctrlKey || e.altKey) return;
   343→      const el = e.target as HTMLElement | null;
   344→      if (
   345→        el &&
   346→        (el.tagName === "INPUT" ||
   347→          el.tagName === "TEXTAREA" ||
   348→          el.isContentEditable)
   349→      )
   350→        return;
   351→      if (e.key === "t") {
   352→        onTheaterChange(!theater);
   353→        return;
   354→      }
   355→      const p = playerRef.current;
   356→      if (!p) return;
   357→      try {
   358→        if (e.key === " " || e.key === "k") {
   359→          e.preventDefault();
   360→          if (p.getPlayerState?.() === 1) p.pauseVideo();
   361→          else p.playVideo();
   362→        } else if (e.key === "ArrowRight") {
   363→          e.preventDefault();
   364→          p.seekTo(p.getCurrentTime() + 5, true);
   365→        } else if (e.key === "ArrowLeft") {
   366→          e.preventDefault();
   367→          p.seekTo(Math.max(0, p.getCurrentTime() - 5), true);
   368→        } else if (e.key === "m") {
   369→          if (p.isMuted?.()) p.unMute();
   370→          else p.mute();
   371→        } else if (e.key === "f") {
   372→          toggleFs();
   373→        }
   374→      } catch {
   375→        /* not ready */
   376→      }
   377→    };
   378→    window.addEventListener("keydown", onKey);
   379→    return () => window.removeEventListener("keydown", onKey);
   380→  }, [theater, onTheaterChange, playerRef]);
   381→
   382→  return (
   383→    <>
   384→      <div className="mx-auto flex max-w-[1700px] flex-wrap items-center gap-3 px-4 pb-3 pt-4">
   385→        <Button variant="ghost" size="sm" onClick={onBack} className="gap-2">
   386→          <ArrowLeft className="h-4 w-4" /> Back
   387→        </Button>
   388→        {idx >= 0 && (
   389→          <span className="text-sm text-muted-foreground">
   390→            {idx + 1} / {queue.length} in queue
   391→          </span>
   392→        )}
   393→        <div className="ml-auto flex items-center gap-3">
   394→          <label className="flex items-center gap-2 text-sm text-muted-foreground">
   395→            <Switch
   396→              checked={autoplay}
   397→              onCheckedChange={onAutoplayChange}
   398→              aria-label="Autoplay next"
   399→            />
   400→            Autoplay
   401→          </label>
   402→          <Button
   403→            variant="outline"
   404→            size="sm"
   405→            onClick={() => onTheaterChange(!theater)}
   406→            aria-pressed={theater}
   407→            title="Theater mode (t)"
   408→            className="gap-2"
   409→          >
   410→            <RectangleHorizontal className="h-4 w-4" /> Theater
   411→          </Button>
   412→          <Button variant="outline" size="sm" onClick={onTv} className="gap-2">
   413→            <Tv className="h-4 w-4" /> TV mode
   414→          </Button>
   415→          {next && (
   416→            <Button
   417→              size="sm"
   418→              onClick={() => selectVideo(next)}
   419→              className="gap-2 bg-[#e5484d] text-white hover:bg-[#d23c41]"
   420→            >
   421→              Next <SkipForward className="h-4 w-4" />
   422→            </Button>
   423→          )}
   424→        </div>
   425→      </div>
   426→
   427→      <div className={theater ? "pb-6" : "mx-auto max-w-[1700px] px-4 pb-4"}>
   428→        {/*
   429→          Theater collapses the two-column grid to a plain block so the player
   430→          bleeds edge-to-edge. The player host div keeps the exact same tree
   431→          position in both modes — swapping only its classes/size — so the
   432→          YouTube iframe is never unmounted and playback continues seamlessly.
   433→        */}
   434→        <div
   435→          className={
   436→            theater
   437→              ? ""
   438→              : "grid gap-6 lg:grid-cols-[minmax(0,1fr)_360px] xl:grid-cols-[minmax(0,1fr)_400px]"
   439→          }
   440→        >
   441→          <div className="min-w-0">
   442→            {(() => {
   443→              const show = (hover || touchUi) && !controlsHidden;
   444→              // Paused (or paused-then-rebuffering): YouTube paints its own
   445→              // pause overlay (center play button, share / watch-later,
   446→              // "More videos" row, logo) over a paused embed and offers no
   447→              // playerVar to turn it off — so a paused embed is never shown;
   448→              // it is covered by the poster + our own play glyph instead.
   449→              const coverPaused =
   450→                pstate === 2 || (pstate === 3 && prevStateRef.current === 2);
   451→              const playedPct = dur ? (scrub ?? cur / dur) * 100 : 0;
   452→              const bufPct = Math.max(playedPct, buf * 100);
   453→              const seekGrad = `linear-gradient(to right, #e5484d 0%, #e5484d ${playedPct}%, rgba(255,255,255,.55) ${playedPct}%, rgba(255,255,255,.55) ${bufPct}%, rgba(255,255,255,.25) ${bufPct}%)`;
   454→              const volPct = muted ? 0 : vol;
   455→              const volGrad = `linear-gradient(to right, #fff 0%, #fff ${volPct}%, rgba(255,255,255,.25) ${volPct}%)`;
   456→              return (
   457→                <div
   458→                  ref={shellRef}
   459→                  className={cn(
   460→                    "relative overflow-hidden bg-black",
   461→                    theater ? "w-full" : "aspect-video w-full rounded-xl"
   462→                  )}
   463→                  style={theater ? { height: "min(85vh, 56.25vw)" } : undefined}
   464→                >
   465→                  <div ref={hostRef} className="h-full w-full" />
   466→
   467→                  {/*
   468→                    Event-capture overlay: the cross-origin YouTube iframe is
   469→                    out-of-process and never reports mouse movement to this
   470→                    page, so a transparent layer above it drives hover
   471→                    detection, click-to-pause and double-click fullscreen.
   472→                    Playback itself is driven through the IFrame API.
   473→                  */}
   474→                  <div
   475→                    className={cn("absolute inset-0", !show && "cursor-none")}
   476→                    onMouseEnter={() => {
   477→                      setHover(true);
   478→                      setControlsHidden(false); // show instantly, no wiggle needed
   479→                      if (pstateRef.current === 1) scheduleHide();
   480→                    }}
   481→                    onMouseMove={onOverlayMouseMove}
   482→                    onMouseLeave={onShellMouseLeave}
   483→                    onClick={onShellClick}
   484→                    onDoubleClick={onShellDoubleClick}
   485→                  >
   486→                    {/* Hover-only control bar — YouTube-style. The bar lives
   487→                        inside the overlay so moving onto it keeps controls up. */}
   488→                    <div
   489→                      data-ct-bar
   490→                      onClick={(e) => e.stopPropagation()}
   491→                      onDoubleClick={(e) => e.stopPropagation()}
   492→                      className={cn(
   493→                        "absolute inset-x-0 bottom-0 z-10 bg-gradient-to-t from-black/85 via-black/40 to-transparent pb-1 pt-8 transition-opacity duration-200",
   494→                        show ? "opacity-100" : "pointer-events-none opacity-0"
   495→                      )}
   496→                    >
   497→                    <div className="px-3">
   498→                      <input
   499→                        type="range"
   500→                        min={0}
   501→                        max={1000}
   502→                        step={1}
   503→                        aria-label="Seek"
   504→                        className="ct-range h-1 w-full"
   505→                        style={{ background: seekGrad }}
   506→                        value={Math.round((scrub ?? (dur ? cur / dur : 0)) * 1000)}
   507→                        onChange={(e) => {
   508→                          const frac = Number(e.target.value) / 1000;
   509→                          setScrub(frac);
   510→                          seekTo(frac);
   511→                        }}
   512→                        onPointerDown={() => {
   513→                          if (touchRef.current) setTouchUi(true);
   514→                        }}
   515→                        onPointerUp={() => setScrub(null)}
   516→                        onKeyUp={() => setScrub(null)}
   517→                        onBlur={() => setScrub(null)}
   518→                      />
   519→                    </div>
   520→                    <div className="flex items-center gap-0.5 px-1.5 text-white">
   521→                      <button
   522→                        onClick={togglePlay}
   523→                        aria-label={pstate === 1 ? "Pause" : "Play"}
   524→                        className="ct-cbtn"
   525→                      >
   526→                        {pstate === 1 ? (
   527→                          <Pause className="h-5 w-5" />
   528→                        ) : (
   529→                          <Play className="h-5 w-5" />
   530→                        )}
   531→                      </button>
   532→                      <button
   533→                        onClick={toggleMute}
   534→                        aria-label={muted ? "Unmute" : "Mute"}
   535→                        className="ct-cbtn"
   536→                      >
   537→                        {muted || vol === 0 ? (
   538→                          <VolumeX className="h-5 w-5" />
   539→                        ) : vol < 50 ? (
   540→                          <Volume1 className="h-5 w-5" />
   541→                        ) : (
   542→                          <Volume2 className="h-5 w-5" />
   543→                        )}
   544→                      </button>
   545→                      <input
   546→                        type="range"
   547→                        min={0}
   548→                        max={100}
   549→                        step={1}
   550→                        aria-label="Volume"
   551→                        className="ct-range h-1 w-16"
   552→                        style={{ background: volGrad }}
   553→                        value={muted ? 0 : vol}
   554→                        onChange={(e) => changeVol(Number(e.target.value))}
   555→                      />
   556→                      <span className="ml-1.5 text-xs tabular-nums text-white/90">
   557→                        {fmtDur(scrub !== null ? scrub * dur : cur) || "0:00"} /{" "}
   558→                        {fmtDur(dur) || "0:00"}
   559→                      </span>
   560→                      <div className="ml-auto flex items-center gap-0.5">
   561→                        <button
   562→                          onClick={() => onTheaterChange(!theater)}
   563→                          aria-pressed={theater}
