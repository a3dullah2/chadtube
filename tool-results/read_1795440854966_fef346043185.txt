     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useState } from "react";
     4→import {
     5→  Dices,
     6→  History,
     7→  Home,
     8→  Menu,
     9→  Moon,
    10→  Play,
    11→  Search,
    12→  Sun,
    13→  Tv,
    14→  X,
    15→} from "lucide-react";
    16→import { useTheme } from "next-themes";
    17→import { Badge } from "@/components/ui/badge";
    18→import { Button } from "@/components/ui/button";
    19→import { Input } from "@/components/ui/input";
    20→import {
    21→  Select,
    22→  SelectContent,
    23→  SelectItem,
    24→  SelectTrigger,
    25→  SelectValue,
    26→} from "@/components/ui/select";
    27→import { Separator } from "@/components/ui/separator";
    28→import { Skeleton } from "@/components/ui/skeleton";
    29→import { useToast } from "@/hooks/use-toast";
    30→import { TvMode } from "./tv-mode";
    31→import { WatchView } from "./watch-view";
    32→import type { Chan, HistEntry, SortMode, Tier, TubeData, Vid } from "./types";
    33→import {
    34→  chanName,
    35→  fmtAge,
    36→  fmtDur,
    37→  fmtTimeAgo,
    38→  fmtViews,
    39→  metaJoin,
    40→  shuffleSeeded,
    41→  TIER_COLORS,
    42→  TIER_LABELS,
    43→  thumbUrl,
    44→} from "./utils";
    45→
    46→const TIERS: Tier[] = ["S", "A", "B", "C", "D", "★"];
    47→const PAGE = 40;
    48→
    49→type View = "home" | "watch" | "history";
    50→
    51→export function TubeApp() {
    52→  const [data, setData] = useState<TubeData | null>(null);
    53→  const [loadError, setLoadError] = useState(false);
    54→  const [view, setView] = useState<View>("home");
    55→  const [current, setCurrent] = useState<Vid | null>(null);
    56→  const [queue, setQueue] = useState<Vid[]>([]);
    57→  const [q, setQ] = useState("");
    58→  const [tier, setTier] = useState<"all" | Tier>("all");
    59→  const [chan, setChan] = useState<number | null>(null);
    60→  const [sort, setSort] = useState<SortMode>("newest");
    61→  const [seed, setSeed] = useState(1);
    62→  const [visible, setVisible] = useState(PAGE);
    63→  const [drawer, setDrawer] = useState(false);
    64→  const [tv, setTv] = useState(false);
    65→  const [tvQueue, setTvQueue] = useState<Vid[]>([]);
    66→  const [tvStart, setTvStart] = useState(0);
    67→  const [hist, setHist] = useState<HistEntry[]>([]);
    68→  const [autoplay, setAutoplay] = useState(true);
    69→  const [mounted, setMounted] = useState(false);
    70→  const { theme, setTheme } = useTheme();
    71→  const { toast } = useToast();
    72→
    73→  // Data + persisted prefs (async: never setState synchronously inside the effect)
    74→  useEffect(() => {
    75→    let alive = true;
    76→    (async () => {
    77→      await Promise.resolve();
    78→      if (!alive) return;
    79→      try {
    80→        const h = localStorage.getItem("mytube.hist");
    81→        if (h) setHist(JSON.parse(h));
    82→        const a = localStorage.getItem("mytube.autoplay");
    83→        if (a !== null) setAutoplay(a === "1");
    84→      } catch {
    85→        /* noop */
    86→      }
    87→      setMounted(true);
    88→      try {
    89→        const r = await fetch("/api/videos");
    90→        if (!r.ok) throw new Error("bad response");
    91→        const d = (await r.json()) as TubeData;
    92→        if (alive) setData(d);
    93→      } catch {
    94→        if (alive) setLoadError(true);
    95→      }
    96→    })();
    97→    return () => {
    98→      alive = false;
    99→    };
   100→  }, []);
   101→
   102→  const pushHist = useCallback((id: string) => {
   103→    setHist((prev) => {
   104→      const next = [{ id, ts: Date.now() }, ...prev.filter((h) => h.id !== id)].slice(0, 300);
   105→      try {
   106→        localStorage.setItem("mytube.hist", JSON.stringify(next));
   107→      } catch {
   108→        /* noop */
   109→      }
   110→      return next;
   111→    });
   112→  }, []);
   113→
   114→  const clearHist = useCallback(() => {
   115→    setHist([]);
   116→    try {
   117→      localStorage.removeItem("mytube.hist");
   118→    } catch {
   119→      /* noop */
   120→    }
   121→    toast({ title: "Watch history cleared" });
   122→  }, [toast]);
   123→
   124→  const changeAutoplay = useCallback((v: boolean) => {
   125→    setAutoplay(v);
   126→    try {
   127→      localStorage.setItem("mytube.autoplay", v ? "1" : "0");
   128→    } catch {
   129→      /* noop */
   130→    }
   131→  }, []);
   132→
   133→  const channels = data?.channels ?? [];
   134→
   135→  const filtered = useMemo(() => {
   136→    if (!data) return [];
   137→    const ql = q.trim().toLowerCase();
   138→    let out = data.videos.filter((v) => {
   139→      if (tier !== "all" && (data.channels[v.c]?.tier as Tier) !== tier) return false;
   140→      if (chan !== null && v.c !== chan) return false;
   141→      if (ql) {
   142→        const t = v.t.toLowerCase();
   143→        const c = `${data.channels[v.c]?.n ?? ""} ${v.up ?? ""}`.toLowerCase();
   144→        if (!t.includes(ql) && !c.includes(ql)) return false;
   145→      }
   146→      return true;
   147→    });
   148→    // Feed-level dedupe: same video via its channel AND the playlist keeps the
   149→    // channel copy in the mixed feed (the ★ filter still shows the full playlist).
   150→    const seen = new Set<string>();
   151→    out = out.filter((v) => !seen.has(v.id) && seen.add(v.id));
   152→    if (sort === "newest")
   153→      out = [...out].sort((a, b) => (b.ts ?? 0) - (a.ts ?? 0) || b.v - a.v);
   154→    else if (sort === "views") out = [...out].sort((a, b) => b.v - a.v);
   155→    else if (sort === "longest") out = [...out].sort((a, b) => b.d - a.d);
   156→    else if (sort === "shuffle") out = shuffleSeeded(out, seed);
   157→    return out;
   158→  }, [data, q, tier, chan, sort, seed]);
   159→
   160→  const applyFilter = useCallback(() => {
   161→    setVisible(PAGE);
   162→    // Typing in search or changing filters should return to the grid
   163→    setView("home");
   164→  }, []);
   165→
   166→  const openFromGrid = useCallback(
   167→    (v: Vid, list: Vid[]) => {
   168→      setQueue(list);
   169→      setCurrent(v);
   170→      setView("watch");
   171→      pushHist(v.id);
   172→      window.scrollTo({ top: 0 });
   173→    },
   174→    [pushHist]
   175→  );
   176→
   177→  const openInQueue = useCallback(
   178→    (v: Vid) => {
   179→      setCurrent(v);
   180→      pushHist(v.id);
   181→      window.scrollTo({ top: 0 });
   182→    },
   183→    [pushHist]
   184→  );
   185→
   186→  const reshuffle = useCallback(() => {
   187→    setSort("shuffle");
   188→    setSeed(Math.floor(Math.random() * 1e9));
   189→    setVisible(PAGE);
   190→    toast({
   191→      title: "Shuffled",
   192→      description: `${filtered.length.toLocaleString()} videos in a brand-new order`,
   193→    });
   194→  }, [filtered.length, toast]);
   195→
   196→  const openTv = useCallback(
   197→    (startIdx?: number) => {
   198→      const list = view === "watch" && queue.length ? queue : filtered;
   199→      if (!list.length) {
   200→        toast({ title: "Nothing to play", description: "No videos match the current filters." });
   201→        return;
   202→      }
   203→      setTvQueue(list);
   204→      setTvStart(
   205→        startIdx !== undefined && startIdx >= 0
   206→          ? startIdx
   207→          : Math.floor(Math.random() * list.length)
   208→      );
   209→      setTv(true);
   210→    },
   211→    [view, queue, filtered, toast]
   212→  );
   213→
   214→  const goHome = useCallback(() => {
   215→    setView("home");
   216→    setDrawer(false);
   217→  }, []);
   218→
   219→  const toggleTheme = useCallback(() => {
   220→    setTheme(theme === "dark" ? "light" : "dark");
   221→  }, [theme, setTheme]);
   222→
   223→  const histVids = useMemo(() => {
   224→    if (!data) return [];
   225→    return hist
   226→      .map((h) => ({ h, v: data.videos.find((x) => x.id === h.id) }))
   227→      .filter((x): x is { h: HistEntry; v: Vid } => !!x.v)
   228→      .map((x) => ({ entry: x.h, vid: x.v }));
   229→  }, [data, hist]);
   230→
   231→  const navItem = (
   232→    icon: React.ReactNode,
   233→    label: string,
   234→    active: boolean,
   235→    onClick: () => void
   236→  ) => (
   237→    <button
   238→      onClick={onClick}
   239→      className={`flex w-full items-center gap-3 rounded-lg px-3 py-2 text-sm font-medium hover:bg-accent ${
   240→        active ? "bg-accent" : ""
   241→      }`}
   242→    >
   243→      {icon}
   244→      {label}
   245→    </button>
   246→  );
   247→
   248→  const sidebarContent = (onNavigate?: () => void) => (
   249→    <div className="flex h-full flex-col">
   250→      <div className="p-3">
   251→        {navItem(<Home className="h-4 w-4" />, "Home", view === "home", () => {
   252→          setTier("all");
   253→          setChan(null);
   254→          applyFilter();
   255→          goHome();
   256→          onNavigate?.();
   257→        })}
   258→        {navItem(<Tv className="h-4 w-4" />, "TV mode", false, () => {
   259→          openTv();
   260→          onNavigate?.();
   261→        })}
   262→        {navItem(<History className="h-4 w-4" />, "History", view === "history", () => {
   263→          setView("history");
   264→          setDrawer(false);
   265→          onNavigate?.();
   266→        })}
   267→      </div>
   268→      <Separator />
   269→      <div className="flex-1 overflow-y-auto p-3">
   270→        {data ? (
   271→          <>
   272→            <p className="px-2 pb-1 pt-1 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   273→              {channels.length} curated channels
   274→            </p>
   275→            <button
   276→              onClick={() => {
   277→                setChan(null);
   278→                applyFilter();
   279→                setView("home");
   280→                onNavigate?.();
   281→              }}
   282→              className={`mt-1 flex w-full items-center justify-between rounded-lg px-3 py-1.5 text-sm hover:bg-accent ${
   283→                chan === null ? "bg-accent font-medium" : ""
   284→              }`}
   285→            >
   286→              All channels
   287→            </button>
   288→            {TIERS.map((t) => {
   289→              const list = channels
   290→                .map((c, i) => ({ c, i }))
   291→                .filter((x) => x.c.tier === t);
   292→              if (!list.length) return null;
   293→              return (
   294→                <div key={t} className="mt-3">
   295→                  <p className="flex items-center gap-2 px-2 pb-1 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   296→                    <span
   297→                      className="inline-block h-2 w-2 rounded-full"
   298→                      style={{ backgroundColor: TIER_COLORS[t] }}
   299→                    />
   300→                    {t} — {TIER_LABELS[t]}
   301→                  </p>
   302→                  {list.map(({ c, i }) => (
   303→                    <button
   304→                      key={c.cid}
   305→                      onClick={() => {
   306→                        setChan(i);
   307→                        applyFilter();
   308→                        setView("home");
   309→                        onNavigate?.();
   310→                      }}
   311→                      className={`flex w-full items-center justify-between gap-2 rounded-lg px-3 py-1.5 text-sm hover:bg-accent ${
   312→                        chan === i ? "bg-accent font-medium" : ""
   313→                      }`}
   314→                    >
   315→                      <span className="truncate">{c.n}</span>
   316→                      <span className="shrink-0 text-xs text-muted-foreground">
   317→                        {c.count}
   318→                      </span>
   319→                    </button>
   320→                  ))}
   321→                </div>
   322→              );
   323→            })}
   324→          </>
   325→        ) : (
   326→          <div className="space-y-2 p-2">
   327→            {Array.from({ length: 8 }).map((_, i) => (
   328→              <Skeleton key={i} className="h-6 w-full" />
   329→            ))}
   330→          </div>
   331→        )}
   332→      </div>
   333→    </div>
   334→  );
   335→
   336→  const chip = (t: "all" | Tier) => {
   337→    const active = tier === t;
   338→    return (
   339→      <button
   340→        key={t}
   341→        onClick={() => {
   342→          setTier(t);
   343→          applyFilter();
   344→        }}
   345→        className={`flex shrink-0 items-center gap-1.5 rounded-full border px-3 py-1.5 text-sm font-medium transition ${
   346→          active
   347→            ? "border-transparent bg-foreground text-background"
   348→            : "bg-secondary text-secondary-foreground hover:bg-accent"
   349→        }`}
   350→        aria-pressed={active}
   351→      >
   352→        {t !== "all" && (
   353→          <span
   354→            className="inline-block h-2 w-2 rounded-full"
   355→            style={{ backgroundColor: TIER_COLORS[t] }}
   356→          />
   357→        )}
   358→        {t === "all" ? "All" : t}
   359→      </button>
   360→    );
   361→  };
   362→
   363→  return (
   364→    <div className="flex min-h-screen flex-col">
   365→      {/* Header */}
   366→      <header className="sticky top-0 z-50 border-b bg-background/95 backdrop-blur supports-[backdrop-filter]:bg-background/80">
   367→        <div className="flex h-14 items-center gap-2 px-3 sm:px-4">
   368→          <Button
   369→            variant="ghost"
   370→            size="icon"
   371→            className="md:hidden"
   372→            onClick={() => setDrawer(true)}
   373→            aria-label="Open menu"
   374→          >
   375→            <Menu className="h-5 w-5" />
   376→          </Button>
   377→          <button
   378→            className="flex items-center gap-2"
   379→            onClick={goHome}
   380→            aria-label="MyTube home"
   381→          >
   382→            <span className="grid h-7 w-10 place-items-center rounded-md bg-[#e5484d]">
   383→              <Play className="h-4 w-4 fill-white text-white" />
   384→            </span>
   385→            <span className="text-lg font-bold tracking-tight">MyTube</span>
   386→          </button>
   387→          <div className="relative mx-auto hidden max-w-xl flex-1 sm:block">
   388→            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   389→            <Input
   390→              value={q}
   391→              onChange={(e) => {
   392→                setQ(e.target.value);
   393→                applyFilter();
   394→              }}
   395→              placeholder="Search videos and channels…"
   396→              className="rounded-full border-0 bg-secondary pl-9"
   397→              aria-label="Search"
   398→            />
   399→          </div>
   400→          <div className="flex-1 sm:hidden" />
   401→          <div className="flex items-center gap-1">
   402→            <Button
   403→              variant="ghost"
   404→              size="icon"
   405→              onClick={reshuffle}
   406→              aria-label="Shuffle everything"
   407→            >
   408→              <Dices className="h-5 w-5" />
   409→            </Button>
   410→            <Button
   411→              variant="ghost"
   412→              size="icon"
   413→              className="hidden sm:inline-flex"
   414→              onClick={() => setView("history")}
   415→              aria-label="Watch history"
   416→            >
   417→              <History className="h-5 w-5" />
   418→            </Button>
   419→            <Button
   420→              size="sm"
   421→              onClick={() => openTv()}
   422→              className="hidden gap-2 bg-[#e5484d] text-white hover:bg-[#d23c41] sm:inline-flex"
   423→            >
   424→              <Tv className="h-4 w-4" /> TV mode
   425→            </Button>
   426→            <Button
   427→              variant="ghost"
   428→              size="icon"
   429→              className="sm:hidden"
   430→              onClick={() => openTv()}
   431→              aria-label="TV mode"
   432→            >
   433→              <Tv className="h-5 w-5" />
   434→            </Button>
   435→            <Button
   436→              variant="ghost"
   437→              size="icon"
   438→              onClick={toggleTheme}
   439→              aria-label="Toggle theme"
   440→            >
   441→              {mounted && theme === "dark" ? (
   442→                <Sun className="h-5 w-5" />
   443→              ) : (
   444→                <Moon className="h-5 w-5" />
   445→              )}
   446→            </Button>
   447→          </div>
   448→        </div>
   449→        <div className="px-3 pb-2 sm:hidden">
   450→          <div className="relative">
   451→            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   452→            <Input
   453→              value={q}
   454→              onChange={(e) => {
   455→                setQ(e.target.value);
   456→                applyFilter();
   457→              }}
   458→              placeholder="Search…"
   459→              className="rounded-full border-0 bg-secondary pl-9"
   460→              aria-label="Search"
   461→            />
   462→          </div>
   463→        </div>
   464→      </header>
   465→
   466→      <div className="flex flex-1">
   467→        {/* Desktop sidebar */}
   468→        <aside className="sticky top-14 hidden h-[calc(100vh-3.5rem)] w-60 shrink-0 overflow-y-auto border-r md:block">
   469→          {sidebarContent()}
   470→        </aside>
   471→
   472→        {/* Mobile drawer */}
   473→        {drawer && (
   474→          <div className="fixed inset-0 z-[60] md:hidden">
   475→            <div
   476→              className="absolute inset-0 bg-black/50"
   477→              onClick={() => setDrawer(false)}
   478→              aria-hidden="true"
   479→            />
   480→            <div className="absolute inset-y-0 left-0 w-72 border-r bg-background">
   481→              {sidebarContent(() => setDrawer(false))}
   482→            </div>
   483→          </div>
   484→        )}
   485→
   486→        <main className="min-w-0 flex-1">
   487→          {view === "watch" && current && data ? (
   488→            <WatchView
   489→              video={current}
   490→              queue={queue}
   491→              channels={channels}
   492→              autoplay={autoplay}
   493→              onAutoplayChange={changeAutoplay}
   494→              onBack={goHome}
   495→              onSelect={openInQueue}
   496→              onTv={() =>
   497→                openTv(Math.max(0, queue.findIndex((v) => v.id === current.id)))
   498→              }
   499→            />
   500→          ) : view === "history" ? (
   501→            <div className="mx-auto max-w-4xl px-4 py-4">
   502→              <div className="mb-4 flex items-center gap-3">
   503→                <h1 className="text-xl font-bold">Watch history</h1>
   504→                {histVids.length > 0 && (
   505→                  <Button variant="outline" size="sm" onClick={clearHist}>
   506→                    Clear history
   507→                  </Button>
   508→                )}
   509→              </div>
   510→              {histVids.length === 0 ? (
   511→                <p className="py-16 text-center text-muted-foreground">
   512→                  Nothing watched yet — hit the dice and shuffle something up.
   513→                </p>
   514→              ) : (
   515→                <div className="flex flex-col gap-1">
   516→                  {histVids.map(({ entry, vid }) => (
   517→                    <button
   518→                      key={vid.id}
   519→                      onClick={() => openFromGrid(vid, histVids.map((x) => x.vid))}
   520→                      className="flex gap-3 rounded-lg p-2 text-left hover:bg-accent"
   521→                    >
   522→                      <div className="relative aspect-video w-44 shrink-0 overflow-hidden rounded-md bg-muted">
   523→                        <img
   524→                          src={thumbUrl(vid.id)}
   525→                          alt=""
   526→                          loading="lazy"
   527→                          className="h-full w-full object-cover"
   528→                        />
   529→                        {fmtDur(vid.d) && (
   530→                          <span className="absolute bottom-1 right-1 rounded bg-black/80 px-1 text-[10px] text-white">
   531→                            {fmtDur(vid.d)}
   532→                          </span>
   533→                        )}
   534→                      </div>
   535→                      <div className="min-w-0">
   536→                        <div className="line-clamp-2 text-sm font-medium">
   537→                          {vid.t}
   538→                        </div>
   539→                        <div className="mt-0.5 text-xs text-muted-foreground">
   540→                          {metaJoin(chanName(vid, channels), fmtViews(vid.v))}
   541→                          {" · "}watched {fmtTimeAgo(entry.ts)}
   542→                        </div>
   543→                      </div>
   544→                    </button>
   545→                  ))}
   546→                </div>
   547→              )}
   548→            </div>
   549→          ) : (
   550→            <>
   551→              {/* Filter chips */}
   552→              <div className="sticky top-14 z-40 border-b bg-background/95 backdrop-blur">
   553→                <div className="flex items-center gap-2 overflow-x-auto px-4 py-2">
   554→                  {chip("all")}
   555→                  {TIERS.map((t) => chip(t))}
   556→                  {chan !== null && (
   557→                    <Badge
   558→                      variant="outline"
   559→                      className="flex shrink-0 items-center gap-1 rounded-full px-3 py-1.5 text-sm"
   560→                    >
   561→                      {channels[chan]?.n}
   562→                      <button
   563→                        onClick={() => setChan(null)}
   564→                        aria-label="Clear channel filter"
   565→                      >
   566→                        <X className="h-3.5 w-3.5" />
   567→                      </button>
   568→                    </Badge>
   569→                  )}
   570→                  <div className="ml-auto shrink-0 pl-2">
   571→                    <Select value={sort} onValueChange={(v) => { setSort(v as SortMode); applyFilter(); }}>
   572→                      <SelectTrigger
   573→                        size="sm"
   574→                        className="w-[150px] rounded-full"
   575→                        aria-label="Sort videos"
   576→                      >
   577→                        <SelectValue />
   578→                      </SelectTrigger>
   579→                      <SelectContent>
   580→                        <SelectItem value="newest">Newest first</SelectItem>
   581→                        <SelectItem value="curated">By channel order</SelectItem>
   582→                        <SelectItem value="views">Most viewed</SelectItem>
   583→                        <SelectItem value="longest">Longest</SelectItem>
   584→                        <SelectItem value="shuffle">Shuffled</SelectItem>
   585→                      </SelectContent>
   586→                    </Select>
   587→                  </div>
   588→                </div>
   589→              </div>
   590→
   591→              {/* Meta line */}
   592→              <div className="px-4 pt-3 text-sm text-muted-foreground">

   608→              </div>
   609→

   638→                          className="group text-left"

   670→                      );
   671→                    })}
   672→                  </div>
   673→                  {visible < filtered.length && (

   675→                      <Button
   676→                        variant="outline"
   677→                        onClick={() => setVisible((n) => n + PAGE * 2)}
   678→                      >
   679→                        Load more ({(filtered.length - visible).toLocaleString()} remaining)
   680→                      </Button>
   681→                    </div>
   682→                  )}
   683→                </>
   684→              )}
   685→            </>
   686→          )}
   687→        </main>
   688→      </div>
   689→
   690→      {/* Footer */}
   691→      <footer className="mt-auto border-t">
   692→        <div className="flex flex-col gap-1 px-4 py-4 text-xs text-muted-foreground sm:flex-row sm:items-center sm:justify-between">

   696→          </span>
   697→          <span>
   698→            Inspired by Karpathy’s TristanTube · curated from Tristan Hume’s
   699→            2026 channel tier list · data harvested with yt-dlp
   700→          </span>
   701→        </div>
   702→      </footer>
   703→

   705→      {tv && (

   707→          queue={tvQueue}
   708→          channels={channels}
