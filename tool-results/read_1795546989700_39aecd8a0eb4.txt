     1→"use client";
     2→
     3→import { useCallback, useEffect, useRef, useState } from "react";
     4→import {
     5→  ArrowLeft,
     6→  Maximize,
     7→  Minimize,
     8→  Pause,
     9→  Play,
    10→  RectangleHorizontal,
    11→  SkipForward,
    12→  Volume1,
    13→  Volume2,
    14→  VolumeX,
    15→} from "lucide-react";
    16→import { Badge } from "@/components/ui/badge";
    17→import { Button } from "@/components/ui/button";
    18→import { ScrollArea } from "@/components/ui/scroll-area";
    19→import { Switch } from "@/components/ui/switch";
    20→import { cn } from "@/lib/utils";
    21→import { useYTPlayer } from "./use-yt";
    22→import { chanName, fmtAge, fmtDur, fmtViews, metaJoin, TIER_COLORS, thumbUrl } from "./utils";
    23→import type { Chan, Tier, Vid } from "./types";
    24→
    25→interface WatchViewProps {
    26→  video: Vid;
    27→  queue: Vid[];
    28→  channels: Chan[];
    29→  autoplay: boolean;
    30→  theater: boolean;
    31→  onAutoplayChange: (v: boolean) => void;
    32→  onTheaterChange: (v: boolean) => void;
    33→  onBack: () => void;
    34→  onSelect: (v: Vid) => void;
    35→}
    36→
    37→export function WatchView({
    38→  video,
    39→  queue,
    40→  channels,
    41→  autoplay,
    42→  theater,
    43→  onAutoplayChange,
    44→  onTheaterChange,
    45→  onBack,
    46→  onSelect,
    47→}: WatchViewProps) {
    48→  const idx = queue.findIndex((v) => v.id === video.id);
    49→  const next =
    50→    idx >= 0 && idx + 1 < queue.length ? queue[idx + 1] : (queue[0] ?? null);
    51→  const chan = channels[video.c];
    52→  const upNext = idx >= 0 ? queue.slice(idx + 1, idx + 26) : queue.slice(0, 25);
    53→  const endRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    54→  const endIntRef = useRef<ReturnType<typeof setInterval> | null>(null);
    55→  const [countdown, setCountdown] = useState<number | null>(null);
    56→
    57→  // ----- custom player chrome (hover-only controls) -----
    58→  const shellRef = useRef<HTMLDivElement | null>(null);
    59→  const idleTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    60→  const clickTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    61→  const touchRef = useRef(false);
    62→  const pstateRef = useRef(-1);
    63→  const [pstate, setPstate] = useState(-1); // YT player state (-1 unstarted .. 3 buffering)
    64→  const [cur, setCur] = useState(0);
    65→  const [dur, setDur] = useState(0);
    66→  const [buf, setBuf] = useState(0);
    67→  const [muted, setMuted] = useState(false);
    68→  const [vol, setVol] = useState(100);
    69→  const [hover, setHover] = useState(false);
    70→  const [touchUi, setTouchUi] = useState(false);
    71→  const [controlsHidden, setControlsHidden] = useState(false);
    72→  const [fs, setFs] = useState(false);
    73→  const [scrub, setScrub] = useState<number | null>(null); // seek fraction while dragging
    74→
    75→  // Controls follow YouTube's rules: visible while the mouse is over the
    76→  // video (or after a touch tap), hidden the moment it leaves, and during
    77→  // playback they fade out after 3s without mouse movement. All visibility
    78→  // changes are driven by events (player state, mouse) — never by effects.
    79→  // Reset-style: every call clears any pending hide, so continuous mouse
    80→  // movement keeps controls up and only true 3s idleness hides them.
    81→  // (The old guard-against-re-schedule version hid the bar mid-hover because
    82→  // the first-scheduled timeout was never extended by later movement.)
    83→  const scheduleHide = (ms = 3000) => {
    84→    cancelHide();
    85→    idleTimer.current = setTimeout(() => {
    86→      idleTimer.current = null;
    87→      setControlsHidden(true);
    88→    }, ms);
    89→  };
    90→
    91→  const cancelHide = () => {
    92→    if (idleTimer.current) {
    93→      clearTimeout(idleTimer.current);
    94→      idleTimer.current = null;
    95→    }
    96→  };
    97→
    98→  // Auto-advance bookkeeping for the ended overlay. Cancelled whenever the
    99→  // user navigates elsewhere, replays, or leaves the watch page.
   100→  const clearEnd = () => {
   101→    if (endRef.current) {
   102→      clearTimeout(endRef.current);
   103→      endRef.current = null;
   104→    }
   105→    if (endIntRef.current) {
   106→      clearInterval(endIntRef.current);
   107→      endIntRef.current = null;
   108→    }
   109→    setCountdown(null);
   110→  };
   111→
   112→  const replay = () => {
   113→    clearEnd();
   114→    const p = playerRef.current;
   115→    try {
   116→      p?.seekTo(0, true);
   117→      p?.playVideo();
   118→    } catch {
   119→      /* not ready */
   120→    }
   121→  };
   122→
   123→  const playNextNow = () => {
   124→    clearEnd();
   125→    if (next) onSelect(next);
   126→  };
   127→
   128→  const togglePlay = () => {
   129→    const p = playerRef.current;
   130→    if (!p) return;
   131→    try {
   132→      if (p.getPlayerState?.() === 1) p.pauseVideo();
   133→      else p.playVideo();
   134→    } catch {
   135→      /* not ready */
   136→    }
   137→  };
   138→
   139→  const toggleMute = () => {
   140→    const p = playerRef.current;
   141→    if (!p) return;
   142→    try {
   143→      if (p.isMuted?.()) p.unMute();
   144→      else p.mute();
   145→    } catch {
   146→      /* not ready */
   147→    }
   148→  };
   149→
   150→  const changeVol = (v: number) => {
   151→    const p = playerRef.current;
   152→    if (!p) return;
   153→    try {
   154→      p.setVolume(v);
   155→      if (v > 0 && p.isMuted?.()) p.unMute();
   156→    } catch {
   157→      /* not ready */
   158→    }
   159→    setVol(v);
   160→  };
   161→
   162→  const toggleFs = () => {
   163→    const el = shellRef.current;
   164→    if (!el) return;
   165→    if (document.fullscreenElement) document.exitFullscreen().catch(() => {});
   166→    else el.requestFullscreen?.().catch(() => {});
   167→  };
   168→
   169→  const seekTo = (frac: number) => {
   170→    const p = playerRef.current;
   171→    if (!p || !dur) return;
   172→    try {
   173→      p.seekTo(frac * dur, true);
   174→    } catch {
   175→      /* not ready */
   176→    }
   177→  };
   178→
   179→  // Click on the video toggles playback; double-click goes fullscreen.
   180→  // The pending click is cancelled when a double-click lands.
   181→  const onShellClick = () => {
   182→    if (touchRef.current) {
   183→      // Mobile-style: tap toggles the chrome; auto-hide only while playing.
   184→      if (controlsHidden) {
   185→        setControlsHidden(false);
   186→        setTouchUi(true);
   187→        if (pstateRef.current === 1) scheduleHide();
   188→      } else {
   189→        cancelHide();
   190→        setControlsHidden(true);
   191→      }
   192→      return;
   193→    }
   194→    if (clickTimer.current) return;
   195→    clickTimer.current = setTimeout(() => {
   196→      clickTimer.current = null;
   197→      togglePlay();
   198→    }, 250);
   199→  };
   200→
   201→  const onShellDoubleClick = () => {
   202→    if (clickTimer.current) {
   203→      clearTimeout(clickTimer.current);
   204→      clickTimer.current = null;
   205→    }
   206→    toggleFs();
   207→  };
   208→
   209→  const onOverlayMouseMove = (e: React.MouseEvent) => {
   210→    setHover(true);
   211→    const inBar = !!(e.target as HTMLElement | null)?.closest?.("[data-ct-bar]");
   212→    if (inBar) {
   213→      // Interacting with the bar itself: pin controls, never fade mid-drag.
   214→      cancelHide();
   215→      setControlsHidden(false);
   216→    } else if (pstateRef.current === 1) {
   217→      setControlsHidden(false);
   218→      scheduleHide(); // reset on every move -> only true idleness hides
   219→    } else {
   220→      // paused / buffering / ended -> controls stay up while hovered
   221→      cancelHide();
   222→      setControlsHidden(false);
   223→    }
   224→  };
   225→
   226→  const onShellMouseLeave = () => {
   227→    setHover(false);
   228→    setTouchUi(false);
   229→    cancelHide();
   230→    setControlsHidden(true); // mouse off the video -> always hide
   231→  };
   232→
   233→  const { hostRef, playerRef } = useYTPlayer({
   234→    videoId: video.id,
   235→    autoplay: true,
   236→    onState: (s) => {
   237→      pstateRef.current = s;
   238→      setPstate(s);
   239→      if (s === 1) scheduleHide();
   240→      else {
   241→        cancelHide();
   242→        setControlsHidden(false); // paused/buffering/ended -> show while hovered
   243→      }
   244→    },
   245→    onEnd: () => {
   246→      if (autoplay && next) {
   247→        // Our own end card replaces YouTube's end screen; advance after 5s.
   248→        setCountdown(5);
   249→        endIntRef.current = setInterval(() => {
   250→          setCountdown((c) => (c !== null && c > 1 ? c - 1 : null));
   251→        }, 1000);
   252→        endRef.current = setTimeout(() => {
   253→          clearEnd();
   254→          onSelect(next);
   255→        }, 5000);
   256→      }
   257→    },
   258→  });
   259→
   260→  useEffect(
   261→    () => () => {
   262→      if (endRef.current) clearTimeout(endRef.current);
   263→      if (endIntRef.current) clearInterval(endIntRef.current);
   264→    },
   265→    []
   266→  );
   267→
   268→  // A pending auto-advance belongs to the video that just ended — navigating
   269→  // to any other video must cancel it instead of hijacking the navigation.
   270→  useEffect(() => {
   271→    clearEnd();
   272→    // eslint-disable-next-line react-hooks/exhaustive-deps
   273→  }, [video.id]);
   274→
   275→  // Player-state polling + controls visibility machinery.
   276→  useEffect(() => {
   277→    try {
   278→      touchRef.current = window.matchMedia?.("(pointer: coarse)").matches ?? false;
   279→    } catch {
   280→      /* noop */
   281→    }
   282→    const poll = setInterval(() => {
   283→      const p = playerRef.current;
   284→      if (!p || typeof p.getCurrentTime !== "function") return;
   285→      try {
   286→        setCur(p.getCurrentTime() || 0);
   287→        setDur(p.getDuration() || 0);
   288→        setBuf(p.getVideoLoadedFraction() || 0);
   289→        setMuted(!!p.isMuted?.());
   290→        const v = p.getVolume?.();
   291→        if (typeof v === "number") setVol(v);
   292→      } catch {
   293→        /* player mid-swap */
   294→      }
   295→    }, 250);
   296→    const onFs = () => setFs(document.fullscreenElement === shellRef.current);
   297→    document.addEventListener("fullscreenchange", onFs);
   298→    return () => {
   299→      clearInterval(poll);
   300→      document.removeEventListener("fullscreenchange", onFs);
   301→      if (idleTimer.current) clearTimeout(idleTimer.current);
   302→      if (clickTimer.current) clearTimeout(clickTimer.current);
   303→    };
   304→  }, []);
   305→
   306→  // YouTube-style shortcuts (t = theater, f = fullscreen, m = mute,
   307→  // space/k = play-pause, arrows = seek). Inline player calls keep this
   308→  // effect's dependencies to just the theater props.
   309→  useEffect(() => {
   310→    const onKey = (e: KeyboardEvent) => {
   311→      if (e.metaKey || e.ctrlKey || e.altKey) return;
   312→      const el = e.target as HTMLElement | null;
   313→      if (
   314→        el &&
   315→        (el.tagName === "INPUT" ||
   316→          el.tagName === "TEXTAREA" ||
   317→          el.isContentEditable)
   318→      )
   319→        return;
   320→      if (e.key === "t") {
   321→        onTheaterChange(!theater);
   322→        return;
   323→      }
   324→      const p = playerRef.current;
   325→      if (!p) return;
   326→      try {
   327→        if (e.key === " " || e.key === "k") {
   328→          e.preventDefault();
   329→          if (p.getPlayerState?.() === 1) p.pauseVideo();
   330→          else p.playVideo();
   331→        } else if (e.key === "ArrowRight") {
   332→          e.preventDefault();
   333→          p.seekTo(p.getCurrentTime() + 5, true);
   334→        } else if (e.key === "ArrowLeft") {
   335→          e.preventDefault();
   336→          p.seekTo(Math.max(0, p.getCurrentTime() - 5), true);
   337→        } else if (e.key === "m") {
   338→          if (p.isMuted?.()) p.unMute();
   339→          else p.mute();
   340→        } else if (e.key === "f") {
   341→          toggleFs();
   342→        }
   343→      } catch {
   344→        /* not ready */
   345→      }
   346→    };
   347→    window.addEventListener("keydown", onKey);
   348→    return () => window.removeEventListener("keydown", onKey);
   349→  }, [theater, onTheaterChange, playerRef]);
   350→
   351→  return (
   352→    <>
   353→      <div className="mx-auto flex max-w-[1700px] flex-wrap items-center gap-3 px-4 pb-3 pt-4">
   354→        <Button variant="ghost" size="sm" onClick={onBack} className="gap-2">
   355→          <ArrowLeft className="h-4 w-4" /> Back
   356→        </Button>
   357→        {idx >= 0 && (
   358→          <span className="text-sm text-muted-foreground">
   359→            {idx + 1} / {queue.length} in queue
   360→          </span>
   361→        )}
   362→        <div className="ml-auto flex items-center gap-3">
   363→          <label className="flex items-center gap-2 text-sm text-muted-foreground">
   364→            <Switch
   365→              checked={autoplay}
   366→              onCheckedChange={onAutoplayChange}
   367→              aria-label="Autoplay next"
   368→            />
   369→            Autoplay
   370→          </label>
   371→          <Button
   372→            variant="outline"
   373→            size="sm"
   374→            onClick={() => onTheaterChange(!theater)}
   375→            aria-pressed={theater}
   376→            title="Theater mode (t)"
   377→            className="gap-2"
   378→          >
   379→            <RectangleHorizontal className="h-4 w-4" /> Theater
   380→          </Button>
   381→          {next && (
   382→            <Button
   383→              size="sm"
   384→              onClick={() => onSelect(next)}
   385→              className="gap-2 bg-[#e5484d] text-white hover:bg-[#d23c41]"
   386→            >
   387→              Next <SkipForward className="h-4 w-4" />
   388→            </Button>
   389→          )}
   390→        </div>
   391→      </div>
   392→
   393→      <div className={theater ? "pb-6" : "mx-auto max-w-[1700px] px-4 pb-4"}>
   394→        {/*
   395→          Theater collapses the two-column grid to a plain block so the player
   396→          bleeds edge-to-edge. The player host div keeps the exact same tree
   397→          position in both modes — swapping only its classes/size — so the
   398→          YouTube iframe is never unmounted and playback continues seamlessly.
   399→        */}
   400→        <div
   401→          className={
   402→            theater
   403→              ? ""
   404→              : "grid gap-6 lg:grid-cols-[minmax(0,1fr)_360px] xl:grid-cols-[minmax(0,1fr)_400px]"
   405→          }
   406→        >
   407→          <div className="min-w-0">
   408→            {(() => {
   409→              const show = (hover || touchUi) && !controlsHidden;
   410→              const playedPct = dur ? (scrub ?? cur / dur) * 100 : 0;
   411→              const bufPct = Math.max(playedPct, buf * 100);
   412→              const seekGrad = `linear-gradient(to right, #e5484d 0%, #e5484d ${playedPct}%, rgba(255,255,255,.55) ${playedPct}%, rgba(255,255,255,.55) ${bufPct}%, rgba(255,255,255,.25) ${bufPct}%)`;
   413→              const volPct = muted ? 0 : vol;
   414→              const volGrad = `linear-gradient(to right, #fff 0%, #fff ${volPct}%, rgba(255,255,255,.25) ${volPct}%)`;
   415→              return (
   416→                <div
   417→                  ref={shellRef}
   418→                  className={cn(
   419→                    "relative overflow-hidden bg-black",
   420→                    theater ? "w-full" : "aspect-video w-full rounded-xl"
   421→                  )}
   422→                  style={theater ? { height: "min(85vh, 56.25vw)" } : undefined}
   423→                >
   424→                  <div ref={hostRef} className="h-full w-full" />
   425→
   426→                  {/*
   427→                    Event-capture overlay: the cross-origin YouTube iframe is
   428→                    out-of-process and never reports mouse movement to this
   429→                    page, so a transparent layer above it drives hover
   430→                    detection, click-to-pause and double-click fullscreen.
   431→                    Playback itself is driven through the IFrame API.
   432→                  */}
   433→                  <div
   434→                    className={cn("absolute inset-0", !show && "cursor-none")}
   435→                    onMouseEnter={() => {
   436→                      setHover(true);
   437→                      setControlsHidden(false); // show instantly, no wiggle needed
   438→                      if (pstateRef.current === 1) scheduleHide();
   439→                    }}
   440→                    onMouseMove={onOverlayMouseMove}
   441→                    onMouseLeave={onShellMouseLeave}
   442→                    onClick={onShellClick}
   443→                    onDoubleClick={onShellDoubleClick}
   444→                  >
   445→                    {/* Hover-only control bar — YouTube-style. The bar lives
   446→                        inside the overlay so moving onto it keeps controls up. */}
   447→                    <div
   448→                      data-ct-bar
   449→                      onClick={(e) => e.stopPropagation()}
   450→                      onDoubleClick={(e) => e.stopPropagation()}
   451→                      className={cn(
   452→                        "absolute inset-x-0 bottom-0 z-10 bg-gradient-to-t from-black/85 via-black/40 to-transparent pb-1 pt-8 transition-opacity duration-200",
   453→                        show ? "opacity-100" : "pointer-events-none opacity-0"
   454→                      )}
   455→                    >
   456→                    <div className="px-3">
   457→                      <input
   458→                        type="range"
   459→                        min={0}
   460→                        max={1000}
   461→                        step={1}
   462→                        aria-label="Seek"
   463→                        className="ct-range h-1 w-full"
   464→                        style={{ background: seekGrad }}
   465→                        value={Math.round((scrub ?? (dur ? cur / dur : 0)) * 1000)}
   466→                        onChange={(e) => {
   467→                          const frac = Number(e.target.value) / 1000;
   468→                          setScrub(frac);
   469→                          seekTo(frac);
   470→                        }}
   471→                        onPointerDown={() => {
   472→                          if (touchRef.current) setTouchUi(true);
   473→                        }}
   474→                        onPointerUp={() => setScrub(null)}
   475→                        onKeyUp={() => setScrub(null)}
   476→                        onBlur={() => setScrub(null)}
   477→                      />
   478→                    </div>
   479→                    <div className="flex items-center gap-0.5 px-1.5 text-white">
   480→                      <button
   481→                        onClick={togglePlay}
   482→                        aria-label={pstate === 1 ? "Pause" : "Play"}
   483→                        className="ct-cbtn"
   484→                      >
   485→                        {pstate === 1 ? (
   486→                          <Pause className="h-5 w-5" />
   487→                        ) : (
   488→                          <Play className="h-5 w-5" />
   489→                        )}
   490→                      </button>
   491→                      <button
   492→                        onClick={toggleMute}
   493→                        aria-label={muted ? "Unmute" : "Mute"}
   494→                        className="ct-cbtn"
   495→                      >
   496→                        {muted || vol === 0 ? (
   497→                          <VolumeX className="h-5 w-5" />
   498→                        ) : vol < 50 ? (
   499→                          <Volume1 className="h-5 w-5" />
   500→                        ) : (
   501→                          <Volume2 className="h-5 w-5" />
   502→                        )}
   503→                      </button>
   504→                      <input
   505→                        type="range"
   506→                        min={0}
   507→                        max={100}
   508→                        step={1}
   509→                        aria-label="Volume"
   510→                        className="ct-range h-1 w-16"
   511→                        style={{ background: volGrad }}
   512→                        value={muted ? 0 : vol}
   513→                        onChange={(e) => changeVol(Number(e.target.value))}
   514→                      />
   515→                      <span className="ml-1.5 text-xs tabular-nums text-white/90">
   516→                        {fmtDur(scrub !== null ? scrub * dur : cur) || "0:00"} /{" "}
   517→                        {fmtDur(dur) || "0:00"}
   518→                      </span>
   519→                      <div className="ml-auto flex items-center gap-0.5">
   520→                        <button
   521→                          onClick={() => onTheaterChange(!theater)}
   522→                          aria-pressed={theater}
   523→                          title="Theater mode (t)"
   524→                          aria-label="Theater mode"
   525→                          className="ct-cbtn"
   526→                        >
   527→                          <RectangleHorizontal className="h-5 w-5" />
   528→                        </button>
   529→                        <button
   530→                          onClick={toggleFs}
   531→                          aria-label={fs ? "Exit fullscreen" : "Fullscreen"}
   532→                          title="Fullscreen (f)"
   533→                          className="ct-cbtn"
   534→                        >
   535→                          {fs ? (
   536→                            <Minimize className="h-5 w-5" />
   537→                          ) : (
   538→                            <Maximize className="h-5 w-5" />
   539→                          )}
   540→                        </button>
   541→                      </div>
   542→                    </div>
   543→                  </div>
   544→                  </div>
   545→
   546→                  {pstate === 3 && (
   547→                    <div className="pointer-events-none absolute inset-0 grid place-items-center">
   548→                      <div className="h-12 w-12 animate-spin rounded-full border-4 border-white/25 border-t-white" />
   549→                    </div>
   550→                  )}
   551→                  {(pstate === -1 || pstate === 5) && (
   552→                    <div className="pointer-events-none absolute inset-0 grid place-items-center">
   553→                      <div className="grid h-16 w-16 place-items-center rounded-full bg-black/60 text-white">
   554→                        <Play className="h-7 w-7 translate-x-0.5" />
   555→                      </div>
   556→                    </div>
   557→                  )}
   558→
   559→                  {pstate === 2 && (
   560→                    /* Pause veil: whenever an embed pauses, YouTube paints its
   561→                       own share / watch-later / "More videos" panel over the
   562→                       frozen frame — none of the official embed params can
   563→                       turn that off. Cover it with our own quiet play chip
   564→                       instead. This layer exists ONLY while paused; it is
   565→                       gone the instant playback resumes and never appears
   566→                       while a video is opening or playing. */
   567→                    <div className="pointer-events-none absolute inset-0 z-[5] grid place-items-center bg-black/80">
   568→                      <div className="grid h-20 w-20 place-items-center rounded-full bg-black/50 text-white ring-1 ring-white/25 animate-in fade-in duration-200">
   569→                        <Play className="h-9 w-9 translate-x-0.5" />
   570→                      </div>
   571→                    </div>
   572→                  )}
   573→
   574→                  {pstate === 0 && (
   575→                    /* Opaque end card: fully covers YouTube's own end screen
