    26→    26→interface WatchViewProps {
    27→    27→  video: Vid;
    28→    28→  queue: Vid[];
    29→    29→  channels: Chan[];
    30→    30→  autoplay: boolean;
    31→    31→  theater: boolean;
    32→    32→  onAutoplayChange: (v: boolean) => void;
    33→    33→  onTheaterChange: (v: boolean) => void;

    38→    38→
    39→    39→export function WatchView({
    40→    40→  video,
    41→    41→  queue,
    42→    42→  channels,
    43→    43→  autoplay,
    44→    44→  theater,
    45→    45→  onAutoplayChange,
    46→    46→  onTheaterChange,
    47→    47→  onBack,
    48→    48→  onSelect,
    49→    49→  onTv,
    50→    50→}: WatchViewProps) {
    51→    51→  const idx = queue.findIndex((v) => v.id === video.id);
    52→    52→  const next =
    53→    53→    idx >= 0 && idx + 1 < queue.length ? queue[idx + 1] : (queue[0] ?? null);
    54→    54→  const chan = channels[video.c];
    55→    55→  const upNext = idx >= 0 ? queue.slice(idx + 1, idx + 26) : queue.slice(0, 25);
    56→    56→  const endRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    57→    57→  const endIntRef = useRef<ReturnType<typeof setInterval> | null>(null);
    58→    58→  const [countdown, setCountdown] = useState<number | null>(null);
    59→    59→  // True while a video is loading and until its first frame actually plays.
    60→    60→  // The YouTube iframe is REUSED across videos (loadVideoById), and it keeps
    61→    61→  // painting the previous video's last frame — or, if that one ended, its
    62→    62→  // full end screen (share / watch-later / "More videos" / logo) — through
    63→    63→  // the unstarted(-1), cued(5) and buffering(3) states. An opaque cover
    64→    64→  // hides all of that; it starts true so even the very first video opens on
    65→    65→  // a clean loading screen instead of stale iframe content.
    66→    66→  const [swapping, setSwapping] = useState(true);
    67→    67→
    68→    68→  // ----- custom player chrome (hover-only controls) -----
    69→    69→  const shellRef = useRef<HTMLDivElement | null>(null);
    70→    70→  const idleTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    71→    71→  const clickTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
    72→    72→  const touchRef = useRef(false);
    73→    73→  const pstateRef = useRef(-1);
    74→    74→  const prevStateRef = useRef(-1); // player state before the current one
    75→    75→  const playedRef = useRef(false); // current video has actually started playing
    76→    76→  const [pstate, setPstate] = useState(-1); // YT player state (-1 unstarted .. 3 buffering)
    77→    77→  const [cur, setCur] = useState(0);
    78→    78→  const [dur, setDur] = useState(0);
    79→    79→  const [buf, setBuf] = useState(0);
    80→    80→  const [muted, setMuted] = useState(false);
    81→    81→  const [vol, setVol] = useState(100);
    82→    82→  const [hover, setHover] = useState(false);
    83→    83→  const [touchUi, setTouchUi] = useState(false);
    84→    84→  const [controlsHidden, setControlsHidden] = useState(false);
    85→    85→  const [fs, setFs] = useState(false);
    86→    86→  const [scrub, setScrub] = useState<number | null>(null); // seek fraction while dragging
    87→    87→
    88→    88→  // Controls follow YouTube's rules: visible while the mouse is over the
    89→    89→  // video (or after a touch tap), hidden the moment it leaves, and during
    90→    90→  // playback they fade out after 3s without mouse movement. All visibility
    91→    91→  // changes are driven by events (player state, mouse) — never by effects.
    92→    92→  // Reset-style: every call clears any pending hide, so continuous mouse
    93→    93→  // movement keeps controls up and only true 3s idleness hides them.
    94→    94→  // (The old guard-against-re-schedule version hid the bar mid-hover because
    95→    95→  // the first-scheduled timeout was never extended by later movement.)
    96→    96→  const scheduleHide = (ms = 3000) => {
    97→    97→    cancelHide();
    98→    98→    idleTimer.current = setTimeout(() => {
    99→    99→      idleTimer.current = null;
   100→   100→      setControlsHidden(true);
   101→   101→    }, ms);
   102→   102→  };
   103→   103→
   104→   104→  const cancelHide = () => {
   105→   105→    if (idleTimer.current) {
   106→   106→      clearTimeout(idleTimer.current);
   107→   107→      idleTimer.current = null;
   108→   108→    }
   109→   109→  };
   110→   110→
   111→   111→  // Auto-advance bookkeeping for the ended overlay. Cancelled whenever the
   112→   112→  // user navigates elsewhere, replays, or leaves the watch page.
   113→   113→  const clearEnd = () => {
   114→   114→    if (endRef.current) {
   115→   115→      clearTimeout(endRef.current);
   116→   116→      endRef.current = null;
   117→   117→    }
   118→   118→    if (endIntRef.current) {
   119→   119→      clearInterval(endIntRef.current);
   120→   120→      endIntRef.current = null;
   121→   121→    }
   122→   122→    setCountdown(null);
   123→   123→  };
   124→   124→
   125→   125→  const replay = () => {
   126→   126→    clearEnd();
   127→   127→    setSwapping(true); // cover the old end screen while it re-buffers
   128→   128→    const p = playerRef.current;
   129→   129→    try {
   130→   130→      p?.seekTo(0, true);
   131→   131→      p?.playVideo();
   132→   132→    } catch {
   133→   133→      /* not ready */
   134→   134→    }
   135→   135→  };
   136→   136→
   137→   137→  // Navigate to another video. Always raise the swap cover: loadVideoById
   138→   138→  // keeps the old iframe content — end screen if the video had ended, else
   139→   139→  // its last frame — visible until the new video's first frame renders.
   140→   140→  // (The [video.id] effect below raises it too; this covers the same-tick
   141→   141→  // render so the end card never flashes on the way out.)
   142→   142→  const selectVideo = (v: Vid) => {
   143→   143→    setSwapping(true);
   144→   144→    clearEnd();
   145→   145→    onSelect(v);
   146→   146→  };
   147→   147→
   148→   148→  const togglePlay = () => {
   149→   149→    const p = playerRef.current;
   150→   150→    if (!p) return;
   151→   151→    try {
   152→   152→      if (p.getPlayerState?.() === 1) p.pauseVideo();
   153→   153→      else p.playVideo();
   154→   154→    } catch {
   155→   155→      /* not ready */
   156→   156→    }
   157→   157→  };
   158→   158→
   159→   159→  const toggleMute = () => {
   160→   160→    const p = playerRef.current;
   161→   161→    if (!p) return;
   162→   162→    try {
   163→   163→      if (p.isMuted?.()) p.unMute();
   164→   164→      else p.mute();
   165→   165→    } catch {
   166→   166→      /* not ready */
   167→   167→    }
   168→   168→  };
   169→   169→
   170→   170→  const changeVol = (v: number) => {
   171→   171→    const p = playerRef.current;
   172→   172→    if (!p) return;
   173→   173→    try {
   174→   174→      p.setVolume(v);
   175→   175→      if (v > 0 && p.isMuted?.()) p.unMute();
   176→   176→    } catch {
   177→   177→      /* not ready */
   178→   178→    }
   179→   179→    setVol(v);
   180→   180→  };
   181→   181→
   182→   182→  const toggleFs = () => {
   183→   183→    const el = shellRef.current;
   184→   184→    if (!el) return;
   185→   185→    if (document.fullscreenElement) document.exitFullscreen().catch(() => {});
   186→   186→    else el.requestFullscreen?.().catch(() => {});
   187→   187→  };
   188→   188→
   189→   189→  const seekTo = (frac: number) => {
   190→   190→    const p = playerRef.current;
   191→   191→    if (!p || !dur) return;
   192→   192→    try {
   193→   193→      p.seekTo(frac * dur, true);
   194→   194→    } catch {
   195→   195→      /* not ready */
   196→   196→    }
   197→   197→  };
   198→   198→
   199→   199→  // Click on the video toggles playback; double-click goes fullscreen.
   200→   200→  // The pending click is cancelled when a double-click lands.
   201→   201→  const onShellClick = () => {
   202→   202→    if (touchRef.current) {
   203→   203→      // Mobile-style: tap toggles the chrome; auto-hide only while playing.
   204→   204→      if (controlsHidden) {
   205→   205→        setControlsHidden(false);
   206→   206→        setTouchUi(true);
   207→   207→        if (pstateRef.current === 1) scheduleHide();
   208→   208→      } else {
   209→   209→        cancelHide();
   210→   210→        setControlsHidden(true);
   211→   211→      }
   212→   212→      return;
   213→   213→    }
   214→   214→    if (clickTimer.current) return;
   215→   215→    clickTimer.current = setTimeout(() => {
   216→   216→      clickTimer.current = null;
   217→   217→      togglePlay();
   218→   218→    }, 250);
   219→   219→  };
   220→   220→
   221→   221→  const onShellDoubleClick = () => {
   222→   222→    if (clickTimer.current) {
   223→   223→      clearTimeout(clickTimer.current);
   224→   224→      clickTimer.current = null;
   225→   225→    }
   226→   226→    toggleFs();
   227→   227→  };
   228→   228→
   229→   229→  const onOverlayMouseMove = (e: React.MouseEvent) => {
   230→   230→    setHover(true);
   231→   231→    const inBar = !!(e.target as HTMLElement | null)?.closest?.("[data-ct-bar]");
   232→   232→    if (inBar) {
   233→   233→      // Interacting with the bar itself: pin controls, never fade mid-drag.
   234→   234→      cancelHide();
   235→   235→      setControlsHidden(false);
   236→   236→    } else if (pstateRef.current === 1) {
   237→   237→      setControlsHidden(false);
   238→   238→      scheduleHide(); // reset on every move -> only true idleness hides
   239→   239→    } else {
   240→   240→      // paused / buffering / ended -> controls stay up while hovered
   241→   241→      cancelHide();
   242→   242→      setControlsHidden(false);
   243→   243→    }
   244→   244→  };
   245→   245→
   246→   246→  const onShellMouseLeave = () => {
   247→   247→    setHover(false);
   248→   248→    setTouchUi(false);
   249→   249→    cancelHide();
   250→   250→    setControlsHidden(true); // mouse off the video -> always hide
   251→   251→  };
   252→   252→
   253→   253→  const { hostRef, playerRef } = useYTPlayer({
   254→   254→    videoId: video.id,
   255→   255→    autoplay: true,
   256→   256→    onState: (s) => {
   257→   257→      prevStateRef.current = pstateRef.current;
   258→   258→      pstateRef.current = s;
   259→   259→      setPstate(s);
   260→   260→      if (s === 1) {
   261→   261→        playedRef.current = true;
   262→   262→        setSwapping(false); // the new video's first frame is on screen
   263→   263→        scheduleHide();
   264→   264→      } else {
   265→   265→        if (s === 2 && playedRef.current) setSwapping(false);
   266→   266→        cancelHide();
   267→   267→        setControlsHidden(false); // paused/buffering/ended -> show while hovered
   268→   268→      }
   269→   269→    },
   270→   270→    onEnd: () => {
   271→   271→      if (autoplay && next) {
   272→   272→        // Our own end card replaces YouTube's end screen; advance after 5s.
   273→   273→        setCountdown(5);
   274→   274→        endIntRef.current = setInterval(() => {
   275→   275→          setCountdown((c) => (c !== null && c > 1 ? c - 1 : null));
   276→   276→        }, 1000);
   277→   277→        endRef.current = setTimeout(() => {
   278→   278→          clearEnd();
   279→   279→          selectVideo(next);
   280→   280→        }, 5000);
   281→   281→      }
   282→   282→    },
   283→   283→  });
   284→   284→
   285→   285→  useEffect(
   286→   286→    () => () => {
   287→   287→      if (endRef.current) clearTimeout(endRef.current);
   288→   288→      if (endIntRef.current) clearInterval(endIntRef.current);
   289→   289→    },
   290→   290→    []
   291→   291→  );
   292→   292→
   293→   293→  // A pending auto-advance belongs to the video that just ended — navigating
   294→   294→  // to any other video must cancel it instead of hijacking the navigation.
   295→   295→  // Every video change also re-raises the swap cover: the reused iframe keeps
   296→   296→  // showing the old video's end screen / last frame through the -1/5/3 load
   297→   297→  // states until the new one actually plays (onState clears it then; a user
   298→   298→  // pause of an already-started video clears it too).
   299→   299→  useEffect(() => {
   300→   300→    clearEnd();
   301→   301→    playedRef.current = false;
   302→   302→    setSwapping(true);
   303→   303→    // eslint-disable-next-line react-hooks/exhaustive-deps
   304→   304→  }, [video.id]);
   305→   305→
   306→   306→  // Player-state polling + controls visibility machinery.
   307→   307→  useEffect(() => {
   308→   308→    try {
   309→   309→      touchRef.current = window.matchMedia?.("(pointer: coarse)").matches ?? false;
   310→   310→    } catch {
   311→   311→      /* noop */
   312→   312→    }
   313→   313→    const poll = setInterval(() => {
   314→   314→      const p = playerRef.current;
   315→   315→      if (!p || typeof p.getCurrentTime !== "function") return;
   316→   316→      try {
   317→   317→        setCur(p.getCurrentTime() || 0);
   318→   318→        setDur(p.getDuration() || 0);
   319→   319→        setBuf(p.getVideoLoadedFraction() || 0);
   320→   320→        setMuted(!!p.isMuted?.());
   321→   321→        const v = p.getVolume?.();
   322→   322→        if (typeof v === "number") setVol(v);
   323→   323→      } catch {
   324→   324→        /* player mid-swap */
   325→   325→      }
   326→   326→    }, 250);
   327→   327→    const onFs = () => setFs(document.fullscreenElement === shellRef.current);
   328→   328→    document.addEventListener("fullscreenchange", onFs);
   329→   329→    return () => {
   330→   330→      clearInterval(poll);
   331→   331→      document.removeEventListener("fullscreenchange", onFs);
   332→   332→      if (idleTimer.current) clearTimeout(idleTimer.current);
   333→   333→      if (clickTimer.current) clearTimeout(clickTimer.current);
   334→   334→    };
   335→   335→  }, []);
   336→   336→
   337→   337→  // YouTube-style shortcuts (t = theater, f = fullscreen, m = mute,
   338→   338→  // space/k = play-pause, arrows = seek). Inline player calls keep this
   339→   339→  // effect's dependencies to just the theater props.
   340→   340→  useEffect(() => {
   341→   341→    const onKey = (e: KeyboardEvent) => {
   342→   342→      if (e.metaKey || e.ctrlKey || e.altKey) return;
   343→   343→      const el = e.target as HTMLElement | null;
   344→   344→      if (
   345→   345→        el &&
   346→   346→        (el.tagName === "INPUT" ||
   347→   347→          el.tagName === "TEXTAREA" ||
   348→   348→          el.isContentEditable)
   349→   349→      )
   350→   350→        return;
   351→   351→      if (e.key === "t") {
   352→   352→        onTheaterChange(!theater);
   353→   353→        return;
   354→   354→      }
   355→   355→      const p = playerRef.current;
   356→   356→      if (!p) return;
   357→   357→      try {
   358→   358→        if (e.key === " " || e.key === "k") {
   359→   359→          e.preventDefault();
   360→   360→          if (p.getPlayerState?.() === 1) p.pauseVideo();
   361→   361→          else p.playVideo();
   362→   362→        } else if (e.key === "ArrowRight") {
   363→   363→          e.preventDefault();
   364→   364→          p.seekTo(p.getCurrentTime() + 5, true);
   365→   365→        } else if (e.key === "ArrowLeft") {
   366→   366→          e.preventDefault();
   367→   367→          p.seekTo(Math.max(0, p.getCurrentTime() - 5), true);
   368→   368→        } else if (e.key === "m") {
   369→   369→          if (p.isMuted?.()) p.unMute();
   370→   370→          else p.mute();
   371→   371→        } else if (e.key === "f") {
   372→   372→          toggleFs();
   373→   373→        }
   374→   374→      } catch {
   375→   375→        /* not ready */
   376→   376→      }
   377→   377→    };
   378→   378→    window.addEventListener("keydown", onKey);
   379→   379→    return () => window.removeEventListener("keydown", onKey);
   380→   380→  }, [theater, onTheaterChange, playerRef]);
   381→   381→
   382→   382→  return (
   383→   383→    <>
   384→   384→      <div className="mx-auto flex max-w-[1700px] flex-wrap items-center gap-3 px-4 pb-3 pt-4">
   385→   385→        <Button variant="ghost" size="sm" onClick={onBack} className="gap-2">
   386→   386→          <ArrowLeft className="h-4 w-4" /> Back
   387→   387→        </Button>
   388→   388→        {idx >= 0 && (
   389→   389→          <span className="text-sm text-muted-foreground">
   390→   390→            {idx + 1} / {queue.length} in queue
   391→   391→          </span>
   392→   392→        )}
   393→   393→        <div className="ml-auto flex items-center gap-3">
   394→   394→          <label className="flex items-center gap-2 text-sm text-muted-foreground">
   395→   395→            <Switch
   396→   396→              checked={autoplay}
   397→   397→              onCheckedChange={onAutoplayChange}
   398→   398→              aria-label="Autoplay next"
   399→   399→            />
   400→   400→            Autoplay
   401→   401→          </label>
   402→   402→          <Button
   403→   403→            variant="outline"
   404→   404→            size="sm"
   405→   405→            onClick={() => onTheaterChange(!theater)}
   406→   406→            aria-pressed={theater}
   407→   407→            title="Theater mode (t)"
   408→   408→            className="gap-2"
   409→   409→          >
   410→   410→            <RectangleHorizontal className="h-4 w-4" /> Theater
   411→   411→          </Button>
   412→   412→          <Button variant="outline" size="sm" onClick={onTv} className="gap-2">
   413→   413→            <Tv className="h-4 w-4" /> TV mode
   414→   414→          </Button>
   415→   415→          {next && (
   416→   416→            <Button
   417→   417→              size="sm"
   418→   418→              onClick={() => selectVideo(next)}
   419→   419→              className="gap-2 bg-[#e5484d] text-white hover:bg-[#d23c41]"
   420→   420→            >
   421→   421→              Next <SkipForward className="h-4 w-4" />
   422→   422→            </Button>
   423→   423→          )}
   424→   424→        </div>
   425→   425→      </div>
   426→   426→
   427→   427→      <div className={theater ? "pb-6" : "mx-auto max-w-[1700px] px-4 pb-4"}>
   428→   428→        {/*
   429→   429→          Theater collapses the two-column grid to a plain block so the player
   430→   430→          bleeds edge-to-edge. The player host div keeps the exact same tree
   431→   431→          position in both modes — swapping only its classes/size — so the
   432→   432→          YouTube iframe is never unmounted and playback continues seamlessly.
   433→   433→        */}
   434→   434→        <div
   435→   435→          className={
   436→   436→            theater
   437→   437→              ? ""
   438→   438→              : "grid gap-6 lg:grid-cols-[minmax(0,1fr)_360px] xl:grid-cols-[minmax(0,1fr)_400px]"
   439→   439→          }
   440→   440→        >
   441→   441→          <div className="min-w-0">
   442→   442→            {(() => {
   443→   443→              const show = (hover || touchUi) && !controlsHidden;
   444→   444→              // Paused (or paused-then-rebuffering): YouTube paints its own
   445→   445→              // pause overlay (center play button, share / watch-later,
   446→   446→              // "More videos" row, logo) over a paused embed and offers no
   447→   447→              // playerVar to turn it off — so a paused embed is never shown;
   448→   448→              // it is covered by the poster + our own play glyph instead.
   449→   449→              const coverPaused =
   450→   450→                pstate === 2 || (pstate === 3 && prevStateRef.current === 2);
   451→   451→              const playedPct = dur ? (scrub ?? cur / dur) * 100 : 0;
   452→   452→              const bufPct = Math.max(playedPct, buf * 100);
   453→   453→              const seekGrad = `linear-gradient(to right, #e5484d 0%, #e5484d ${playedPct}%, rgba(255,255,255,.55) ${playedPct}%, rgba(255,255,255,.55) ${bufPct}%, rgba(255,255,255,.25) ${bufPct}%)`;
   454→   454→              const volPct = muted ? 0 : vol;
   455→   455→              const volGrad = `linear-gradient(to right, #fff 0%, #fff ${volPct}%, rgba(255,255,255,.25) ${volPct}%)`;
   456→   456→              return (
   457→   457→                <div
   458→   458→                  ref={shellRef}
   459→   459→                  className={cn(
   460→   460→                    "relative overflow-hidden bg-black",
   461→   461→                    theater ? "w-full" : "aspect-video w-full rounded-xl"
   462→   462→                  )}
   463→   463→                  style={theater ? { height: "min(85vh, 56.25vw)" } : undefined}
   464→   464→                >
   465→   465→                  <div ref={hostRef} className="h-full w-full" />
   466→   466→
